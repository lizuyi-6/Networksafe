/**
 * FluxGuard Agent CLI - 命令行启动入口
 * 
 * 用法:
 *   npm run agent                    # 使用默认服务器地址
 *   npm run agent -- --server=http://your-server:8080
 */
import { startAgent } from './agent';
import logger from './utils/logger';
// 解析命令行参数
function parseArgs(): { serverUrl: string } {
    const args = process.argv.slice(2);
    let serverUrl = process.env.FLUXGUARD_SERVER || 'http://localhost:8080';
    for (const arg of args) {
        if (arg.startsWith('--server=')) {
            serverUrl = arg.replace('--server=', '');
        }
    }
    return { serverUrl };
}

import express from 'express';
import cors from 'cors';
import { defaultConfig as config } from './config';
import logger from './utils/logger';
// Import Routes
import statsRoutes from './api/routes/stats.routes';
import configRoutes from './api/routes/config.routes';
import logsRoutes from './api/routes/logs.routes';
import sitesRoutes from './api/routes/sites.routes';
import adminRoutes from './api/routes/admin.routes';
const app = express();
// ============================================
// 安全中间件配置 (符合 OWASP 规范)
// ============================================
// 1. CORS 配置 - 限制允许的源
const allowedOrigins = process.env.ALLOWED_ORIGINS?.split(',') || [
    'http://localhost:5173',
    'http://127.0.0.1:5173',
    'https://localhost:5173',
];
app.use(cors({
    origin: (origin, callback) => {
        // 允许无 origin 的请求（如服务器间请求）
        if (!origin) return callback(null, true);
        if (allowedOrigins.includes(origin)) {
            callback(null, true);
        } else {
            logger.warn(`Blocked CORS request from: ${origin}`);
            callback(new Error('Not allowed by CORS'));
        }
    },
    credentials: true,
    methods: ['GET', 'POST', 'PUT', 'DELETE', 'OPTIONS'],
    allowedHeaders: ['Content-Type', 'Authorization', 'X-Requested-With'],
    maxAge: 86400, // 预检请求缓存 24 小时
}));
// 2. 请求体大小限制 (防止大请求攻击)
app.use(express.json({ limit: '10kb' }));
app.use(express.urlencoded({ extended: true, limit: '10kb' }));
// 3. 安全 HTTP 头 (手动实现，或安装 helmet)
app.use((req, res, next) => {
    // 防止点击劫持
    res.setHeader('X-Frame-Options', 'DENY');
    // 防止 MIME 类型嗅探
    res.setHeader('X-Content-Type-Options', 'nosniff');
    // XSS 过滤
    res.setHeader('X-XSS-Protection', '1; mode=block');
    // 隐藏服务器信息
    res.removeHeader('X-Powered-By');
    // HSTS (仅 HTTPS)
    if (req.secure) {
        res.setHeader('Strict-Transport-Security', 'max-age=31536000; includeSubDomains');
    }
    next();
});
// 4. 请求日志
app.use((req, res, next) => {
    logger.info(`${req.method} ${req.path}`, {
        ip: req.ip,
        userAgent: req.headers['user-agent']?.substring(0, 50),
    });
    next();
});
// Routes
app.use('/api/stats', statsRoutes);
app.use('/api/config', configRoutes);
app.use('/api/logs', logsRoutes);
app.use('/api/sites', sitesRoutes);
app.use('/api/admin', adminRoutes);
// Health check
app.get('/health', (req, res) => {
    res.status(200).json({ status: 'ok', uptime: process.uptime() });
});

app.use((err: any, req: express.Request, res: express.Response, next: express.NextFunction) => {

    logger.error('Unhandled error', { error: err.message, stack: err.stack });
    res.status(500).json({ error: 'Internal Server Error' });
});
export default app;

// File: agent.ts
/**
 * FluxGuard Agent - 轻量级安全防护代理
 * 
 * 功能:
 * - 心跳上报
 * - 流量监控
 * - 攻击检测
 * - 防火墙控制
 * - 与中央管控平台通信
 */
import os from 'os';
import { io, Socket } from 'socket.io-client';
import logger from '../utils/logger';
import { TrafficMonitor } from './traffic-monitor';
import { AttackDetector } from './attack-detector';
import { FirewallController } from './firewall-controller';
export interface AgentConfig {
    serverUrl: string;       // 中央管控平台地址
    agentId?: string;        // Agent 唯一标识 (自动生成)
    heartbeatInterval?: number; // 心跳间隔 (ms)
}
export interface AgentStatus {
    agentId: string;
    hostname: string;
    platform: string;
    uptime: number;
    cpuUsage: number;
    memoryUsage: number;
    lastHeartbeat: string;
    isOnline: boolean;
}
export class Agent {
    private config: AgentConfig;
    private socket: Socket | null = null;
    private agentId: string;
    private heartbeatTimer: NodeJS.Timeout | null = null;
    private trafficMonitor: TrafficMonitor;
    private attackDetector: AttackDetector;
    private firewallController: FirewallController;
    constructor(config: AgentConfig) {
        this.config = {
            heartbeatInterval: 30000, // 默认 30 秒
            ...config
        };
        // 生成唯一 Agent ID (基于主机名 + MAC 地址)
        this.agentId = config.agentId || this.generateAgentId();
        // 初始化子模块
        this.trafficMonitor = new TrafficMonitor();
        this.attackDetector = new AttackDetector();
        this.firewallController = new FirewallController();
    }
    /**
     * 生成唯一 Agent ID
     */
    private generateAgentId(): string {
        const hostname = os.hostname();
        const interfaces = os.networkInterfaces();
        let mac = 'unknown';
        for (const name of Object.keys(interfaces)) {
            for (const iface of interfaces[name] || []) {
                if (!iface.internal && iface.mac !== '00:00:00:00:00:00') {
                    mac = iface.mac.replace(/:/g, '');
                    break;
                }
            }
        }
        return `agent-${hostname}-${mac.substring(0, 8)}`;
    }
    /**
     * 启动 Agent
     */
    async start(): Promise<void> {
        logger.info(`Starting FluxGuard Agent: ${this.agentId}`);
        // 连接到中央管控平台
        await this.connectToServer();
        // 启动心跳
        this.startHeartbeat();
        // 启动流量监控
        await this.trafficMonitor.start((event) => {
            // 分析流量事件
            const threat = this.attackDetector.analyze(event);
            if (threat) {
                this.handleThreat(threat);
            }
        });
        logger.info('Agent started successfully');
    }
    /**
     * 连接到中央管控平台
     */
    private async connectToServer(): Promise<void> {
        return new Promise((resolve, reject) => {
            logger.info(`Connecting to server: ${this.config.serverUrl}`);
            this.socket = io(this.config.serverUrl, {
                auth: { agentId: this.agentId },
                reconnection: true,
                reconnectionAttempts: Infinity,
                reconnectionDelay: 5000
            });
            this.socket.on('connect', () => {
                logger.info('Connected to control platform');
                this.registerAgent();
                resolve();
            });
            this.socket.on('disconnect', () => {
                logger.warn('Disconnected from control platform');
            });
            this.socket.on('connect_error', (error) => {
                logger.error('Connection error', { error: error.message });
            });
            // 监听来自平台的指令
            this.socket.on('command', (command) => {
                this.handleCommand(command);
            });
            // 设置连接超时
            setTimeout(() => {
                if (!this.socket?.connected) {
                    logger.warn('Connection timeout, will retry in background');
                    resolve(); // 不阻塞启动
                }
            }, 10000);
        });
    }
    /**
     * 向平台注册 Agent
     */
    private registerAgent(): void {
        if (!this.socket) return;
        const registration = {
            agentId: this.agentId,
            hostname: os.hostname(),
            platform: os.platform(),
            arch: os.arch(),
            cpus: os.cpus().length,
            totalMemory: os.totalmem(),
            version: '1.0.0'
        };
        this.socket.emit('agent:register', registration);
        logger.info('Agent registered with platform');
    }
    /**
     * 启动心跳
     */
    private startHeartbeat(): void {
        this.heartbeatTimer = setInterval(() => {
            this.sendHeartbeat();
        }, this.config.heartbeatInterval);
        // 立即发送一次
        this.sendHeartbeat();
    }
    /**
     * 发送心跳
     */
    private sendHeartbeat(): void {
        if (!this.socket?.connected) return;
        const status: AgentStatus = {
            agentId: this.agentId,
            hostname: os.hostname(),
            platform: os.platform(),
            uptime: os.uptime(),
            cpuUsage: this.getCpuUsage(),
            memoryUsage: this.getMemoryUsage(),
            lastHeartbeat: new Date().toISOString(),
            isOnline: true
        };
        this.socket.emit('agent:heartbeat', status);
    }
    /**
     * 获取 CPU 使用率
     */
    private getCpuUsage(): number {
        const cpus = os.cpus();
        let totalIdle = 0;
        let totalTick = 0;
        for (const cpu of cpus) {
            for (const type in cpu.times) {
                totalTick += cpu.times[type as keyof typeof cpu.times];
            }
            totalIdle += cpu.times.idle;
        }
        return Math.round((1 - totalIdle / totalTick) * 100);
    }
    /**
     * 获取内存使用率
     */
    private getMemoryUsage(): number {
        const total = os.totalmem();
        const free = os.freemem();
        return Math.round((1 - free / total) * 100);
    }
    /**
     * 处理来自平台的指令
     */
    private async handleCommand(command: { type: string; payload?: any }): Promise<void> {
        logger.info(`Received command: ${command.type}`);
        switch (command.type) {
            case 'scan':
                // 触发本地扫描
                await this.runLocalScan();
                break;
            case 'block_ip':
                // 封禁 IP
                if (command.payload?.ip) {
                    await this.firewallController.blockIP(command.payload.ip);
                }
                break;
            case 'unblock_ip':
                // 解封 IP
                if (command.payload?.ip) {
                    await this.firewallController.unblockIP(command.payload.ip);
                }
                break;
            case 'update_rules':
                // 更新检测规则
                if (command.payload?.rules) {
                    this.attackDetector.updateRules(command.payload.rules);
                }
                break;
            default:
                logger.warn(`Unknown command: ${command.type}`);
        }
    }
    /**
     * 处理检测到的威胁
     */
    private handleThreat(threat: any): void {
        logger.warn('Threat detected', { threat });
        // 上报威胁到平台
        if (this.socket?.connected) {
            this.socket.emit('agent:threat', {
                agentId: this.agentId,
                threat,
                timestamp: new Date().toISOString()
            });
        }
        // 自动阻断 (如果配置了)
        if (threat.autoBlock && threat.sourceIP) {
            this.firewallController.blockIP(threat.sourceIP, threat.blockDuration || 3600);
        }
    }
    /**
     * 运行本地扫描
     */
    private async runLocalScan(): Promise<void> {
        logger.info('Running local security scan...');
        // 这里可以调用现有的 Scanner 模块
        // 结果通过 WebSocket 上报
        if (this.socket?.connected) {
            this.socket.emit('agent:scan_started', { agentId: this.agentId });
        }
        // TODO: 集成现有 Scanner
    }
    /**
     * 停止 Agent
     */
    async stop(): Promise<void> {
        logger.info('Stopping Agent...');
        if (this.heartbeatTimer) {
            clearInterval(this.heartbeatTimer);
        }
        await this.trafficMonitor.stop();
        if (this.socket) {
            this.socket.disconnect();
        }
        logger.info('Agent stopped');
    }
}

/**
 * 攻击检测引擎
 * 
 * 实时分析流量，检测各类攻击模式：

export interface ThreatAlert {

                ],

            }
        ];
        logger.info(`Loaded ${this.rules.length} detection rules`);
    }
    /**
     * 分析流量事件
     */
    analyze(event: TrafficEvent): ThreatAlert | null {
        // 更新 IP 请求计数
        this.updateIPCounter(event.sourceIP);
        // 检查是否为 DDoS/暴力破解
        const rateThreat = this.checkRateLimit(event.sourceIP);
        if (rateThreat) {
            return rateThreat;
        }
        // 构建待检测的内容
        const contentToCheck = [
            event.path || '',
            event.requestBody || '',
            event.userAgent || '',
            event.rawLog || ''
        ].join(' ');
        // 遍历规则进行检测
        for (const rule of this.rules) {
            for (const pattern of rule.patterns) {
                if (pattern.test(contentToCheck)) {
                    const threat: ThreatAlert = {
                        type: rule.type,
                        severity: rule.severity,
                        sourceIP: event.sourceIP,
                        description: rule.name,
                        evidence: this.extractEvidence(contentToCheck, pattern),
                        ruleId: rule.id,
                        autoBlock: rule.autoBlock,
                        blockDuration: rule.blockDuration,
                        timestamp: new Date()
                    };
                    logger.warn(`Attack detected: ${rule.name}`, {
                        ip: event.sourceIP,
                        ruleId: rule.id
                    });
                    return threat;
                }
            }
        }
        return null;
    }
    /**
     * 更新 IP 请求计数器
     */
    private updateIPCounter(ip: string): void {
        const now = Date.now();
        const existing = this.ipRequestCount.get(ip);
        if (existing) {
            existing.count++;
        } else {
            this.ipRequestCount.set(ip, { count: 1, firstSeen: now });
        }
    }
    /**
     * 检查请求速率限制
     */
    private checkRateLimit(ip: string): ThreatAlert | null {
        const data = this.ipRequestCount.get(ip);
        if (!data) return null;
        const elapsed = (Date.now() - data.firstSeen) / 1000; // 秒
        const ratePerSecond = data.count / Math.max(elapsed, 1);
        // 超过 100 请求/秒 视为 DDoS
        if (ratePerSecond > 100) {
            return {
                type: 'ddos',
                severity: 'critical',
                sourceIP: ip,
                description: 'DDoS Attack Detected - High Request Rate',
                evidence: `${data.count} requests in ${elapsed.toFixed(1)}s (${ratePerSecond.toFixed(1)}/s)`,
                ruleId: 'DDOS-001',
                autoBlock: true,
                blockDuration: 3600,
                timestamp: new Date()
            };
        }
        // 超过 30 请求/秒 视为暴力破解
        if (ratePerSecond > 30) {
            return {
                type: 'brute_force',
                severity: 'high',
                sourceIP: ip,
                description: 'Possible Brute Force Attack',
                evidence: `${data.count} requests in ${elapsed.toFixed(1)}s (${ratePerSecond.toFixed(1)}/s)`,
                ruleId: 'BF-001',
                autoBlock: true,
                blockDuration: 1800,
                timestamp: new Date()
            };
        }
        return null;
    }
    /**
     * 提取攻击证据
     */
    private extractEvidence(content: string, pattern: RegExp): string {
        const match = content.match(pattern);
        if (match) {
            // 返回匹配内容前后各 20 个字符
            const index = content.indexOf(match[0]);
            const start = Math.max(0, index - 20);
            const end = Math.min(content.length, index + match[0].length + 20);
            return `...${content.substring(start, end)}...`;
        }
        return '';
    }
    /**
     * 启动清理计时器
     */
    private startCleanupTimer(): void {
        // 每分钟清理过期的计数器
        this.cleanupInterval = setInterval(() => {
            const now = Date.now();
            const expiryTime = 60000; // 1 分钟
            for (const [ip, data] of this.ipRequestCount) {
                if (now - data.firstSeen > expiryTime) {
                    this.ipRequestCount.delete(ip);
                }
            }
        }, 60000);
    }
    /**
     * 更新检测规则
     */
    updateRules(newRules: DetectionRule[]): void {
        this.rules = newRules;
        logger.info(`Updated detection rules: ${newRules.length} rules loaded`);
    }
    /**
     * 添加自定义规则
     */
    addRule(rule: DetectionRule): void {
        this.rules.push(rule);
        logger.info(`Added detection rule: ${rule.id}`);
    }
    /**
     * 获取所有规则
     */
    getRules(): DetectionRule[] {
        return [...this.rules];
    }
    /**
     * 停止检测器
     */
    stop(): void {
        if (this.cleanupInterval) {
            clearInterval(this.cleanupInterval);
        }
        this.ipRequestCount.clear();
    }
}

/**
 * 防火墙控制器
 * 
 * 调用系统防火墙工具封禁/解封 IP:
 * - Linux: iptables / firewalld / ufw
 * - Windows: netsh
 */
import { exec } from 'child_process';
import { promisify } from 'util';
import os from 'os';
import logger from '../utils/logger';
const execAsync = promisify(exec);
interface BlockedIP {
    ip: string;
    reason: string;
    blockedAt: Date;
    expiresAt?: Date;
    timerId?: NodeJS.Timeout;
}
type FirewallBackend = 'iptables' | 'firewalld' | 'ufw' | 'netsh' | 'none';
export class FirewallController {
    private blockedIPs: Map<string, BlockedIP> = new Map();
    private backend: FirewallBackend = 'none';
    constructor() {
        this.detectBackend();
    }
    /**
     * 检测可用的防火墙后端
     */
    private async detectBackend(): Promise<void> {
        const platform = os.platform();
        if (platform === 'win32') {
            this.backend = 'netsh';
            logger.info('Firewall backend: Windows netsh');
            return;
        }
        // Linux: 按优先级检测
        try {
            await execAsync('which firewall-cmd');
            this.backend = 'firewalld';
            logger.info('Firewall backend: firewalld');
            return;
        } catch { }
        try {
            await execAsync('which ufw');
            this.backend = 'ufw';
            logger.info('Firewall backend: ufw');
            return;
        } catch { }
        try {
            await execAsync('which iptables');
            this.backend = 'iptables';
            logger.info('Firewall backend: iptables');
            return;
        } catch { }
        logger.warn('No firewall backend detected');
        this.backend = 'none';
    }
    /**
     * 封禁 IP
     * @param ip 要封禁的 IP 地址
     * @param duration 封禁时长 (秒), 0 表示永久
     * @param reason 封禁原因
     */
    async blockIP(ip: string, duration: number = 3600, reason: string = 'Threat detected'): Promise<boolean> {
        // 验证 IP 格式
        if (!this.isValidIP(ip)) {
            logger.error(`Invalid IP address: ${ip}`);
            return false;
        }
        // 跳过本地地址
        if (this.isLocalIP(ip)) {
            logger.warn(`Skipping local IP: ${ip}`);
            return false;
        }
        // 检查是否已封禁
        if (this.blockedIPs.has(ip)) {
            logger.info(`IP already blocked: ${ip}`);
            return true;
        }
        let success = false;
        try {
            switch (this.backend) {
                case 'iptables':
                    success = await this.blockWithIptables(ip);
                    break;
                case 'firewalld':
                    success = await this.blockWithFirewalld(ip);
                    break;
                case 'ufw':
                    success = await this.blockWithUfw(ip);
                    break;
                case 'netsh':
                    success = await this.blockWithNetsh(ip);
                    break;
                default:
                    logger.warn(`No firewall backend available, IP not blocked: ${ip}`);
                    // 即使没有防火墙，也记录到内存
                    success = true;
            }
            if (success) {
                const blockedIP: BlockedIP = {
                    ip,
                    reason,
                    blockedAt: new Date(),
                    expiresAt: duration > 0 ? new Date(Date.now() + duration * 1000) : undefined
                };
                // 设置自动解封计时器
                if (duration > 0) {
                    blockedIP.timerId = setTimeout(() => {
                        this.unblockIP(ip);
                    }, duration * 1000);
                }
                this.blockedIPs.set(ip, blockedIP);
                logger.info(`Blocked IP: ${ip} for ${duration}s (${reason})`);
            }
            return success;
        } catch (error: any) {
            logger.error(`Failed to block IP: ${ip}`, { error: error.message });
            return false;
        }
    }
    /**
     * 解封 IP
     */
    async unblockIP(ip: string): Promise<boolean> {
        const blocked = this.blockedIPs.get(ip);
        if (!blocked) {
            logger.info(`IP not in block list: ${ip}`);
            return true;
        }
        // 清除计时器
        if (blocked.timerId) {
            clearTimeout(blocked.timerId);
        }
        let success = false;
        try {
            switch (this.backend) {
                case 'iptables':
                    success = await this.unblockWithIptables(ip);
                    break;
                case 'firewalld':
                    success = await this.unblockWithFirewalld(ip);
                    break;
                case 'ufw':
                    success = await this.unblockWithUfw(ip);
                    break;
                case 'netsh':
                    success = await this.unblockWithNetsh(ip);
                    break;
                default:
                    success = true;
            }
            if (success) {
                this.blockedIPs.delete(ip);
                logger.info(`Unblocked IP: ${ip}`);
            }
            return success;
        } catch (error: any) {
            logger.error(`Failed to unblock IP: ${ip}`, { error: error.message });
            return false;
        }
    }
    // === iptables 实现 ===
    private async blockWithIptables(ip: string): Promise<boolean> {
        try {
            await execAsync(`iptables -A INPUT -s ${ip} -j DROP`);
            return true;
        } catch (error: any) {
            logger.error(`iptables block failed: ${error.message}`);
            return false;
        }
    }
    private async unblockWithIptables(ip: string): Promise<boolean> {
        try {
            await execAsync(`iptables -D INPUT -s ${ip} -j DROP`);
            return true;
        } catch (error: any) {
            logger.error(`iptables unblock failed: ${error.message}`);
            return false;
        }
    }
    // === firewalld 实现 ===
    private async blockWithFirewalld(ip: string): Promise<boolean> {
        try {
            await execAsync(`firewall-cmd --add-rich-rule='rule family="ipv4" source address="${ip}" reject'`);
            return true;
        } catch (error: any) {
            logger.error(`firewalld block failed: ${error.message}`);
            return false;
        }
    }
    private async unblockWithFirewalld(ip: string): Promise<boolean> {
        try {
            await execAsync(`firewall-cmd --remove-rich-rule='rule family="ipv4" source address="${ip}" reject'`);
            return true;
        } catch (error: any) {
            logger.error(`firewalld unblock failed: ${error.message}`);
            return false;
        }
    }
    // === ufw 实现 ===
    private async blockWithUfw(ip: string): Promise<boolean> {
        try {
            await execAsync(`ufw deny from ${ip}`);
            return true;
        } catch (error: any) {
            logger.error(`ufw block failed: ${error.message}`);
            return false;
        }
    }
    private async unblockWithUfw(ip: string): Promise<boolean> {
        try {
            await execAsync(`ufw delete deny from ${ip}`);
            return true;
        } catch (error: any) {
            logger.error(`ufw unblock failed: ${error.message}`);
            return false;
        }
    }
    // === Windows netsh 实现 ===
    private async blockWithNetsh(ip: string): Promise<boolean> {
        try {
            const ruleName = `FluxGuard_Block_${ip.replace(/\./g, '_')}`;
            await execAsync(`netsh advfirewall firewall add rule name="${ruleName}" dir=in action=block remoteip=${ip}`);
            return true;
        } catch (error: any) {
            logger.error(`netsh block failed: ${error.message}`);
            return false;
        }
    }
    private async unblockWithNetsh(ip: string): Promise<boolean> {
        try {
            const ruleName = `FluxGuard_Block_${ip.replace(/\./g, '_')}`;
            await execAsync(`netsh advfirewall firewall delete rule name="${ruleName}"`);
            return true;
        } catch (error: any) {
            logger.error(`netsh unblock failed: ${error.message}`);
            return false;
        }
    }
    // === 辅助方法 ===
    /**
     * 验证 IP 格式
     */
    private isValidIP(ip: string): boolean {
        const ipv4Regex = /^(\d{1,3}\.){3}\d{1,3}$/;
        const ipv6Regex = /^([0-9a-fA-F]{1,4}:){7}[0-9a-fA-F]{1,4}$/;
        return ipv4Regex.test(ip) || ipv6Regex.test(ip);
    }
    /**
     * 检查是否为本地 IP
     */
    private isLocalIP(ip: string): boolean {
        return ip === '127.0.0.1' ||
            ip === '::1' ||
            ip.startsWith('192.168.') ||
            ip.startsWith('10.') ||
            ip.startsWith('172.16.');
    }
    /**
     * 获取所有被封禁的 IP
     */
    getBlockedIPs(): BlockedIP[] {
        return Array.from(this.blockedIPs.values());
    }
    /**
     * 检查 IP 是否被封禁
     */
    isBlocked(ip: string): boolean {
        return this.blockedIPs.has(ip);
    }
    /**
     * 获取当前使用的防火墙后端
     */
    getBackend(): FirewallBackend {
        return this.backend;
    }
    /**
     * 清除所有封禁
     */
    async clearAll(): Promise<void> {
        for (const ip of this.blockedIPs.keys()) {
            await this.unblockIP(ip);
        }
    }
}

    private async watchLogFile(logPath: string, source: string): Promise<void> {
        try {
            // 获取文件当前大小（从末尾开始读取新内容）
            const stats = fs.statSync(logPath);
            this.filePositions.set(logPath, stats.size);
            // 监控文件变化
            const watcher = fs.watch(logPath, (eventType) => {
                if (eventType === 'change') {
                    this.readNewEntries(logPath, source);
                }
            });
            this.watchers.set(logPath, watcher);
            logger.info(`Watching log file: ${logPath} (${source})`);
        } catch (error: any) {
            logger.error(`Failed to watch log file: ${logPath}`, { error: error.message });
        }
    }
    /**
     * 读取新的日志条目
     */
    private readNewEntries(logPath: string, source: string): void {
        try {
            const stats = fs.statSync(logPath);
            const currentPosition = this.filePositions.get(logPath) || 0;
            // 检测日志轮转
            if (stats.size < currentPosition) {
                this.filePositions.set(logPath, 0);
                return;
            }
            if (stats.size === currentPosition) {
                return; // 无新内容
            }
            // 读取新内容
            const stream = fs.createReadStream(logPath, {
                start: currentPosition,
                end: stats.size
            });
            const rl = readline.createInterface({ input: stream });
            rl.on('line', (line) => {
                const entry = this.parseLogLine(line, source);
                if (entry) {
                    this.processEntry(entry);
                }
            });
            this.filePositions.set(logPath, stats.size);
        } catch (error: any) {
            logger.error(`Failed to read log entries: ${logPath}`, { error: error.message });
        }
    }
    /**
     * 解析日志行
     */
    private parseLogLine(line: string, source: string): LogEntry | null {
        if (!line.trim()) return null;
        // 尝试解析时间戳
        let timestamp = new Date();
        let level: LogEntry['level'] = 'info';
        let message = line;
        // Syslog 格式: "Jan 24 10:30:45 hostname service: message"
        const syslogMatch = line.match(/^(\w{3}\s+\d{1,2}\s+\d{2}:\d{2}:\d{2})\s+(\S+)\s+(\S+?):\s*(.*)$/);
        if (syslogMatch) {
            const [, timeStr, , , msg] = syslogMatch;
            timestamp = new Date(`${new Date().getFullYear()} ${timeStr}`);
            message = msg;
        }
        // 检测日志级别
        const lowerLine = line.toLowerCase();
        if (lowerLine.includes('error') || lowerLine.includes('fail')) {
            level = 'error';
        } else if (lowerLine.includes('warn')) {
            level = 'warn';
        } else if (lowerLine.includes('critical') || lowerLine.includes('fatal') || lowerLine.includes('panic')) {
            level = 'critical';
        } else if (lowerLine.includes('debug')) {
            level = 'debug';
        }
        return {
            source,
            timestamp,
            level,
            message,
            raw: line
        };
    }
    /**
     * 处理日志条目
     */
    private processEntry(entry: LogEntry): void {
        // 发送通用日志事件
        this.emit('log', entry);
        // 检测危险模式
        for (const pattern of this.dangerousPatterns) {
            const match = entry.raw.match(pattern.pattern);
            if (match) {
                const alert = {
                    type: 'log_pattern',
                    pattern: pattern.name,
                    severity: pattern.severity,
                    description: pattern.description,
                    entry,
                    match: match[0],
                    timestamp: new Date()
                };
                logger.warn(`Dangerous pattern detected: ${pattern.name}`, {
                    source: entry.source,
                    message: entry.message.substring(0, 100)
                });
                this.emit('alert', alert);
            }
        }
        // 发送按级别分类的事件
        if (entry.level === 'error' || entry.level === 'critical') {
            this.emit('error', entry);
        }
    }
    /**
     * 添加自定义检测模式
     */
    addPattern(pattern: LogPattern): void {
        this.dangerousPatterns.push(pattern);
        logger.info(`Added log pattern: ${pattern.name}`);
    }
    /**
     * 手动添加日志文件监控
     */
    async addLogFile(logPath: string, source: string): Promise<boolean> {
        if (!fs.existsSync(logPath)) {
            logger.warn(`Log file not found: ${logPath}`);
            return false;
        }
        if (this.watchers.has(logPath)) {
            logger.warn(`Already watching: ${logPath}`);
            return false;
        }
        await this.watchLogFile(logPath, source);
        return true;
    }
    /**
     * 获取当前监控的文件列表
     */
    getWatchedFiles(): string[] {
        return Array.from(this.watchers.keys());
    }
    /**
     * 停止日志采集
     */
    stop(): void {
        this.isRunning = false;
        for (const [path, watcher] of this.watchers) {
            watcher.close();
            logger.info(`Stopped watching: ${path}`);
        }
        this.watchers.clear();
        this.filePositions.clear();
        logger.info('Log collector stopped');
    }
}

/**
 * 流量监控模块
 * 
 * 监控服务器入站流量，分析请求模式
 */
import { exec } from 'child_process';
import { promisify } from 'util';
import fs from 'fs';
import readline from 'readline';
import logger from '../utils/logger';
const execAsync = promisify(exec);
export interface TrafficEvent {
    timestamp: Date;
    sourceIP: string;
    destPort: number;
    method?: string;
    path?: string;
    userAgent?: string;
    requestBody?: string;
    rawLog?: string;
}
export type TrafficCallback = (event: TrafficEvent) => void;
export class TrafficMonitor {
    private isRunning = false;
    private callback: TrafficCallback | null = null;
    private logWatcher: fs.FSWatcher | null = null;
    private logStream: readline.Interface | null = null;
    private lastLogPosition = 0;
    // 常见日志路径
    private readonly LOG_PATHS = {
        nginx: ['/var/log/nginx/access.log', '/var/log/nginx/error.log'],
        apache: ['/var/log/apache2/access.log', '/var/log/httpd/access_log'],
        custom: process.env.ACCESS_LOG_PATH
    };
    /**
     * 启动流量监控
     */
    async start(callback: TrafficCallback): Promise<void> {
        if (this.isRunning) {
            logger.warn('Traffic monitor already running');
            return;
        }
        this.callback = callback;
        this.isRunning = true;
        // 尝试找到可用的日志文件
        const logPath = await this.findAccessLog();
        if (logPath) {
            await this.watchLogFile(logPath);
        } else {
            logger.warn('No access log found, using netstat monitoring');
            await this.fallbackToNetstat();
        }
        logger.info('Traffic monitoring started');
    }
    /**
     * 查找可用的访问日志
     */
    private async findAccessLog(): Promise<string | null> {
        // 优先使用自定义路径
        if (this.LOG_PATHS.custom && fs.existsSync(this.LOG_PATHS.custom)) {
            return this.LOG_PATHS.custom;
        }
        // 检查 Nginx 日志
        for (const path of this.LOG_PATHS.nginx) {
            if (fs.existsSync(path)) {
                logger.info(`Found Nginx log: ${path}`);
                return path;
            }
        }
        // 检查 Apache 日志
        for (const path of this.LOG_PATHS.apache) {
            if (fs.existsSync(path)) {
                logger.info(`Found Apache log: ${path}`);
                return path;
            }
        }
        return null;
    }
    /**
     * 监控日志文件
     */
    private async watchLogFile(logPath: string): Promise<void> {
        logger.info(`Watching log file: ${logPath}`);
        // 获取当前文件位置 (只读取新内容)
        const stats = fs.statSync(logPath);
        this.lastLogPosition = stats.size;
        // 使用 fs.watch 监控文件变化
        this.logWatcher = fs.watch(logPath, (eventType) => {
            if (eventType === 'change') {
                this.readNewLogEntries(logPath);
            }
        });
    }
    /**
     * 读取新的日志条目
     */
    private readNewLogEntries(logPath: string): void {
        const stats = fs.statSync(logPath);
        if (stats.size < this.lastLogPosition) {
            // 日志被轮转，从头开始
            this.lastLogPosition = 0;
        }
        if (stats.size === this.lastLogPosition) {
            return; // 无新内容
        }
        const stream = fs.createReadStream(logPath, {
            start: this.lastLogPosition,
            end: stats.size
        });
        const rl = readline.createInterface({ input: stream });
        rl.on('line', (line) => {
            const event = this.parseLogLine(line);
            if (event && this.callback) {
                this.callback(event);
            }
        });
        this.lastLogPosition = stats.size;
    }
    /**
     * 解析日志行
     */
    private parseLogLine(line: string): TrafficEvent | null {
        // 尝试解析 Nginx/Apache Combined Log Format
        // 格式: IP - - [timestamp] "METHOD PATH HTTP/1.1" status size "referer" "user-agent"
        const combinedLogRegex = /^(\S+) \S+ \S+ \[([^\]]+)\] "(\S+) (\S+) \S+" (\d+) \S+ "[^"]*" "([^"]*)"/;
        const match = line.match(combinedLogRegex);
        if (match) {
            return {
                timestamp: new Date(),
                sourceIP: match[1],
                destPort: 80, // 从日志无法直接获取端口
                method: match[3],
                path: match[4],
                userAgent: match[6],
                rawLog: line
            };
        }
        // 尝试解析其他格式
        // TODO: 支持更多日志格式
        return null;
    }
    /**
     * 备用方案：使用 netstat 监控
     */
    private async fallbackToNetstat(): Promise<void> {
        // 定期执行 netstat 获取连接信息
        const checkConnections = async () => {
            if (!this.isRunning) return;
            try {
                const { stdout } = await execAsync('netstat -tn 2>/dev/null || ss -tn');
                const connections = this.parseNetstatOutput(stdout);
                for (const conn of connections) {
                    if (this.callback) {
                        this.callback(conn);
                    }
                }
            } catch (error) {
                // 静默处理错误
            }
            // 每 5 秒检查一次
            if (this.isRunning) {
                setTimeout(checkConnections, 5000);
            }
        };
        checkConnections();
    }
    /**
     * 解析 netstat 输出
     */
    private parseNetstatOutput(output: string): TrafficEvent[] {
        const events: TrafficEvent[] = [];
        const lines = output.split('\n');
        for (const line of lines) {
            // 匹配 ESTABLISHED 连接
            const match = line.match(/(\d+\.\d+\.\d+\.\d+):(\d+)\s+(\d+\.\d+\.\d+\.\d+):(\d+)/);
            if (match) {
                events.push({
                    timestamp: new Date(),
                    sourceIP: match[1],
                    destPort: parseInt(match[4])
                });
            }
        }
        return events;
    }
    /**
     * 停止监控
     */
    async stop(): Promise<void> {
        this.isRunning = false;
        if (this.logWatcher) {
            this.logWatcher.close();
            this.logWatcher = null;
        }
        if (this.logStream) {
            this.logStream.close();
            this.logStream = null;
        }
        logger.info('Traffic monitoring stopped');
    }
}

// Password validation
const PASSWORD_MIN_LENGTH = 8;
const EMAIL_REGEX = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
/**
 * POST /api/auth/register
 * Register a new user
 */
router.post('/register', async (req: AuthenticatedRequest, res: Response) => {
    try {
        const { email, password, username } = req.body;
        // Validate input
        if (!email || !password || !username) {
            return res.status(400).json({
                error: 'Email, password, and username are required',
                code: 'VALIDATION_ERROR'
            });
        }
        if (!EMAIL_REGEX.test(email)) {
            return res.status(400).json({
                error: 'Invalid email format',
                code: 'INVALID_EMAIL'
            });
        }
        if (password.length < PASSWORD_MIN_LENGTH) {
            return res.status(400).json({
                error: `Password must be at least ${PASSWORD_MIN_LENGTH} characters`,
                code: 'PASSWORD_TOO_SHORT'
            });
        }
        const db = getDatabase();
        // Check if user already exists
        const existingUser = db.prepare('SELECT id FROM users WHERE email = ?').get(email);
        if (existingUser) {
            return res.status(409).json({
                error: 'Email already registered',
                code: 'USER_EXISTS'
            });
        }
        // Hash password
        const saltRounds = 12;
        const passwordHash = await bcrypt.hash(password, saltRounds);
        // Create user
        const userId = uuidv4();
        const tenantId = uuidv4(); // Each user gets their own tenant by default
        const role = 'tenant'; // Default role
        db.prepare(`
      INSERT INTO users (id, email, username, password_hash, role, tenant_id, created_at)
      VALUES (?, ?, ?, ?, ?, ?, datetime('now'))
    `).run(userId, email, username, passwordHash, role, tenantId);
        // Generate tokens
        const tokens = generateTokens({
            userId,
            email,
            username,
            role,
            tenantId
        });
        logger.info(`New user registered: ${email}`);
        res.status(201).json({
            message: 'User registered successfully',
            user: {
                userId,
                email,
                username,
                role,
                tenantId
            },
            ...tokens
        });
    } catch (error: any) {
        logger.error('Registration error', { error: error.message });
        res.status(500).json({
            error: 'Registration failed',
            code: 'INTERNAL_ERROR'
        });
    }
});

        if (!email || !password) {
            return res.status(400).json({
                error: 'Email and password are required',
                code: 'VALIDATION_ERROR'
            });
        }
        const db = getDatabase();
        // Find user
        const user = db.prepare(`
      SELECT id, email, username, password_hash, role, tenant_id, status
      FROM users WHERE email = ?
    `).get(email) as any;
        if (!user) {
            return res.status(401).json({
                error: 'Invalid credentials',
                code: 'INVALID_CREDENTIALS'
            });
        }
        // Check if account is active
        if (user.status === 'suspended') {
            return res.status(403).json({
                error: 'Account suspended',
                code: 'ACCOUNT_SUSPENDED'
            });
        }
        // Verify password
        const passwordValid = await bcrypt.compare(password, user.password_hash);

            logger.warn(`Failed login attempt for: ${email}`);
            return res.status(401).json({
                error: 'Invalid credentials',
                code: 'INVALID_CREDENTIALS'
            });
        }
        // Update last login
        db.prepare(`UPDATE users SET last_login = datetime('now') WHERE id = ?`).run(user.id);
        // Generate tokens
        const tokens = generateTokens({
            userId: user.id,
            email: user.email,
            username: user.username,
            role: user.role,
            tenantId: user.tenant_id
        });
        logger.info(`User logged in: ${email}`);
        res.json({
            message: 'Login successful',
            user: {
                userId: user.id,
                email: user.email,
                username: user.username,

                tenantId: user.tenant_id
            },

        });
    }

            });

 * GET /api/sites

 */

    const db = getDatabase().getDb();
    const sites = db.prepare('SELECT * FROM sites ORDER BY createdAt DESC').all();
    // Convert integer booleans
    const formattedSites = sites.map((site: any) => ({
      ...site,
      sslEnabled: Boolean(site.sslEnabled),
      antiBotEnabled: Boolean(site.antiBotEnabled),
      rateLimitEnabled: Boolean(site.rateLimitEnabled),
      concurrencyEnabled: Boolean(site.concurrencyEnabled)
    }));
    res.json({ success: true, data: formattedSites });
  } catch (error: any) {
    logger.error('Failed to get sites', { error });
    res.status(500).json({ success: false, error: error.message });
  }
});
/**

 * Get site by ID

router.get('/:id', async (req: Request, res: Response) => {
  try {
    const db = getDatabase().getDb();

      return res.status(404).json({ success: false, error: 'Site not found' });
    }
    res.json({ success: true, data: site });
  } catch (error: any) {
    logger.error('Failed to get site', { error });
    res.status(500).json({ success: false, error: error.message });
  }
});
/**
 * POST /api/sites
 * Create new site
 */
router.post('/', async (req: Request, res: Response) => {
  try {
    const { domain, target, sslEnabled = true, security } = req.body;
    if (!domain || !target) {
      return res.status(400).json({ success: false, error: 'domain and target are required' });
    }
    const db = getDatabase().getDb();
    const stmt = db.prepare(`
      INSERT INTO sites (
        domain, target, sslEnabled,
        antiBotEnabled, rateLimitEnabled, concurrencyEnabled,
        rateLimitRequests, concurrencyMax
      ) VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    `);
    const result = stmt.run(
      domain,
      target,
      sslEnabled ? 1 : 0,
      security?.antiBot ? 1 : 0,
      security?.rateLimit?.enabled ? 1 : 0,
      security?.concurrency?.enabled ? 1 : 0,
      security?.rateLimit?.requestsPerMinute || 60,
      security?.concurrency?.maxConcurrent || 5
    );
    logger.info(`Site created: ${domain} -> ${target}`);
    res.json({
      success: true,
      data: { id: result.lastInsertRowid, domain, target }
    });
  } catch (error: any) {
    logger.error('Failed to create site', { error });
    res.status(500).json({ success: false, error: error.message });
  }
});
/**
 * PUT /api/sites/:id
 * Update site
 */
router.put('/:id', async (req: Request, res: Response) => {
  try {
    const { domain, target, sslEnabled, security } = req.body;
    const db = getDatabase().getDb();
    const stmt = db.prepare(`
      UPDATE sites SET
        domain = COALESCE(?, domain),
        target = COALESCE(?, target),
        sslEnabled = COALESCE(?, sslEnabled),
        antiBotEnabled = COALESCE(?, antiBotEnabled),
        rateLimitEnabled = COALESCE(?, rateLimitEnabled),
        concurrencyEnabled = COALESCE(?, concurrencyEnabled),
        rateLimitRequests = COALESCE(?, rateLimitRequests),
        concurrencyMax = COALESCE(?, concurrencyMax),
        updatedAt = CURRENT_TIMESTAMP
      WHERE id = ?
    `);
    stmt.run(
      domain,
      target,
      sslEnabled !== undefined ? (sslEnabled ? 1 : 0) : null,
      security?.antiBot !== undefined ? (security.antiBot ? 1 : 0) : null,
      security?.rateLimit?.enabled !== undefined ? (security.rateLimit.enabled ? 1 : 0) : null,
      security?.concurrency?.enabled !== undefined ? (security.concurrency.enabled ? 1 : 0) : null,
      security?.rateLimit?.requestsPerMinute || null,
      security?.concurrency?.maxConcurrent || null,
      req.params.id
    );
    logger.info(`Site updated: ${req.params.id}`);
    res.json({ success: true });
  } catch (error: any) {
    logger.error('Failed to update site', { error });
    res.status(500).json({ success: false, error: error.message });
  }
});

import { Router, Request, Response } from 'express';
import { SmartFixer } from '../../scanner/smart-fixer';
import { OptimizationEngine } from '../../scanner/optimization-engine';
import logger from '../../utils/logger';
const router = Router();
const optimizationEngine = new OptimizationEngine();
/**
 * POST /api/scanner/smart-fix
 * Fix a problem using AI
 */
router.post('/smart-fix', async (req: Request, res: Response) => {
  try {
    const { recommendation, context, options } = req.body;
    if (!recommendation) {
      return res.status(400).json({ success: false, error: 'recommendation is required' });
    }
    logger.info(`Starting smart fix for: ${recommendation.title}`);
    // Create smart fixer with options
    const smartFixer = new SmartFixer({
      enableAI: options?.enableAI !== false,
      aiProvider: options?.aiProvider || {
        name: process.env.AI_PROVIDER || 'ollama',
        apiKey: process.env.OPENAI_API_KEY || process.env.ANTHROPIC_API_KEY,
        baseURL: process.env.AI_BASE_URL,
        model: process.env.AI_MODEL || 'llama2'
      },
      maxAttempts: options?.maxAttempts || 3,
      executionTimeout: options?.executionTimeout || 60000,
      enableRealTimeMonitoring: options?.enableRealTimeMonitoring !== false
    });
    // Run smart fix
    const result = await smartFixer.fixProblem(recommendation, context);
    res.json({
      success: true,
      data: result
    });
  } catch (error: any) {
    logger.error('Smart fix failed', { error });
    res.status(500).json({ success: false, error: error.message });
  }
});
/**
 * POST /api/scanner/batch-fix
 * Fix multiple problems in batch
 */
router.post('/batch-fix', async (req: Request, res: Response) => {
  try {
    const { recommendations, context, options } = req.body;
    if (!recommendations || !Array.isArray(recommendations)) {
      return res.status(400).json({ success: false, error: 'recommendations array is required' });
    }
    logger.info(`Batch fixing ${recommendations.length} problems`);
    const smartFixer = new SmartFixer(options);
    const results = await smartFixer.batchFix(recommendations, context);
    const summary = {
      total: results.length,
      successful: results.filter(r => r.success).length,
      failed: results.filter(r => !r.success).length,
      totalAttempts: results.reduce((sum, r) => sum + r.attempts.length, 0),
      totalTime: results.reduce((sum, r) => sum + r.totalTime, 0)
    };
    logger.info('Batch fix complete', { summary });
    res.json({
      success: true,
      data: {
        results,
        summary
      }
    });
  } catch (error: any) {
    logger.error('Batch fix failed', { error });
    res.status(500).json({ success: false, error: error.message });
  }
});

import initSqlJs, { Database as SqlJsDatabase } from 'sql.js';
import path from 'path';
import fs from 'fs';
import logger from '../utils/logger';
/**
 * Database Manager
 * Handles SQLite database operations using sql.js (pure JavaScript)
 */
export class DatabaseManager {
  private db: SqlJsDatabase | null = null;
  private dbPath: string;
  private saveInterval: NodeJS.Timeout | null = null;
  constructor() {
    const dataDir = process.env.DATA_DIR || path.join(process.cwd(), 'data');
    this.dbPath = path.join(dataDir, 'newsecurity.db');
    this.ensureDataDir(dataDir);
  }
  /**
   * Initialize database (async because sql.js needs to load WASM)
   */
  public async init(): Promise<void> {
    try {
      const SQL = await initSqlJs();
      // Try to load existing database
      if (fs.existsSync(this.dbPath)) {
        const fileBuffer = fs.readFileSync(this.dbPath);
        this.db = new SQL.Database(fileBuffer);
        logger.info(`Loaded existing database: ${this.dbPath}`);
      } else {
        this.db = new SQL.Database();
        logger.info(`Created new database: ${this.dbPath}`);
      }
      this.initSchema();
      this.startAutoSave();
    } catch (error: any) {
      logger.error('Failed to initialize database', { error: error.message });
      throw error;
    }
  }
  /**
   * Ensure data directory exists
   */
  private ensureDataDir(dir: string): void {
    if (!fs.existsSync(dir)) {
      fs.mkdirSync(dir, { recursive: true });
      logger.info(`Created data directory: ${dir}`);
    }
  }
  /**
   * Initialize database schema
   */
  private initSchema(): void {
    if (!this.db) {
      throw new Error('Database not connected');
    }
    try {
      // Sites table
      this.db.run(`
        CREATE TABLE IF NOT EXISTS sites (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          domain TEXT UNIQUE NOT NULL,
          target TEXT NOT NULL,
          sslEnabled INTEGER DEFAULT 1,
          antiBotEnabled INTEGER DEFAULT 1,
          rateLimitEnabled INTEGER DEFAULT 1,
          rateLimitRequests INTEGER DEFAULT 60,
          concurrencyEnabled INTEGER DEFAULT 1,
          concurrencyMax INTEGER DEFAULT 5,
          createdAt DATETIME DEFAULT CURRENT_TIMESTAMP,
          updatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
        )
      `);
      // Security rules table
      this.db.run(`
        CREATE TABLE IF NOT EXISTS security_rules (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          siteId INTEGER NOT NULL,
          ruleType TEXT NOT NULL,
          config TEXT NOT NULL,
          enabled INTEGER DEFAULT 1,
          createdAt DATETIME DEFAULT CURRENT_TIMESTAMP,
          FOREIGN KEY (siteId) REFERENCES sites(id) ON DELETE CASCADE
        )
      `);
      // Scan results table
      this.db.run(`
        CREATE TABLE IF NOT EXISTS scan_results (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          siteId INTEGER,
          scanType TEXT NOT NULL,
          result TEXT NOT NULL,
          recommendations TEXT,
          status TEXT DEFAULT 'pending',
          createdAt DATETIME DEFAULT CURRENT_TIMESTAMP,
          FOREIGN KEY (siteId) REFERENCES sites(id) ON DELETE CASCADE
        )
      `);
      // Backups table
      this.db.run(`
        CREATE TABLE IF NOT EXISTS backups (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          name TEXT NOT NULL,
          description TEXT,
          data TEXT NOT NULL,
          size INTEGER,
          isAutomatic INTEGER DEFAULT 0,
          createdAt DATETIME DEFAULT CURRENT_TIMESTAMP
        )
      `);
      // Monitoring data table (time-series)
      this.db.run(`
        CREATE TABLE IF NOT EXISTS metrics (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          siteId INTEGER NOT NULL,
          metricType TEXT NOT NULL,
          value REAL NOT NULL,
          metadata TEXT,
          timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
          FOREIGN KEY (siteId) REFERENCES sites(id) ON DELETE CASCADE
        )
      `);
      // Agents table
      this.db.run(`
        CREATE TABLE IF NOT EXISTS agents (
          id TEXT PRIMARY KEY,
          hostname TEXT NOT NULL,
          platform TEXT,
          arch TEXT,
          version TEXT,
          lastHeartbeat DATETIME,
          status TEXT DEFAULT 'offline',
          createdAt DATETIME DEFAULT CURRENT_TIMESTAMP
        )
      `);
      // Users table (for authentication)
      this.db.run(`
        CREATE TABLE IF NOT EXISTS users (
          id TEXT PRIMARY KEY,
          email TEXT UNIQUE NOT NULL,
          username TEXT NOT NULL,
          password_hash TEXT NOT NULL,
          role TEXT DEFAULT 'viewer' CHECK(role IN ('admin', 'tenant', 'viewer')),
          tenant_id TEXT,
          status TEXT DEFAULT 'active' CHECK(status IN ('active', 'suspended', 'pending')),
          last_login DATETIME,
          created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
          updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
      `);
      // API Keys table (for agent/device authentication)
      this.db.run(`
        CREATE TABLE IF NOT EXISTS api_keys (
          id TEXT PRIMARY KEY,
          name TEXT NOT NULL,
          key_hash TEXT UNIQUE NOT NULL,
          user_id TEXT,
          tenant_id TEXT,
          permissions TEXT DEFAULT '["read"]',
          last_used DATETIME,
          expires_at DATETIME,
          created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
          FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
        )
      `);
      // Devices table (for ESP32 dongles)
      this.db.run(`
        CREATE TABLE IF NOT EXISTS devices (
          id TEXT PRIMARY KEY,
          device_type TEXT DEFAULT 'esp32',
          mac_address TEXT UNIQUE,
          firmware_version TEXT,
          status TEXT DEFAULT 'offline',
          last_seen DATETIME,
          config TEXT,
          created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
      `);
      // Alerts table (for security alerts)
      this.db.run(`
        CREATE TABLE IF NOT EXISTS alerts (
          id INTEGER PRIMARY KEY AUTOINCREMENT,
          type TEXT NOT NULL,
          severity TEXT DEFAULT 'info' CHECK(severity IN ('info', 'warning', 'critical')),
          source TEXT,
          message TEXT NOT NULL,
          details TEXT,
          acknowledged INTEGER DEFAULT 0,
          acknowledged_by TEXT,
          acknowledged_at DATETIME,
          created_at DATETIME DEFAULT CURRENT_TIMESTAMP
        )
      `);
      // Create indexes
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_sites_domain ON sites(domain)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_scan_results_siteId ON scan_results(siteId)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_metrics_siteId ON metrics(siteId)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_metrics_timestamp ON metrics(timestamp)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_users_email ON users(email)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_users_tenant ON users(tenant_id)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_api_keys_user ON api_keys(user_id)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_alerts_severity ON alerts(severity)`);
      this.db.run(`CREATE INDEX IF NOT EXISTS idx_alerts_created ON alerts(created_at)`);
      logger.info('Database schema initialized');
    } catch (error: any) {
      logger.error('Failed to initialize database schema', { error: error.message });
      throw error;
    }
  }
  /**
   * Start auto-save interval
   */
  private startAutoSave(): void {
    // Save to disk every 30 seconds
    this.saveInterval = setInterval(() => {
      this.save();
    }, 30000);
  }
  /**
   * Save database to disk
   */
  public save(): void {
    if (!this.db) return;
    try {
      const data = this.db.export();
      const buffer = Buffer.from(data);
      fs.writeFileSync(this.dbPath, buffer);
    } catch (error: any) {
      logger.error('Failed to save database', { error: error.message });
    }
  }
  /**
   * Get database instance
   */
  public getDb(): SqlJsDatabase {
    if (!this.db) {
      throw new Error('Database not connected');
    }
    return this.db;
  }
  /**
   * Execute SQL and return results
   */
  public exec(sql: string): void {
    if (!this.db) throw new Error('Database not connected');
    this.db.run(sql);
  }
  /**
   * Prepare and run a statement
   */
  public prepare(sql: string): {
    run: (...params: any[]) => void;
    all: (...params: any[]) => any[];
    get: (...params: any[]) => any;
  } {
    if (!this.db) throw new Error('Database not connected');
    const db = this.db;
    return {
      run: (...params: any[]) => {
        db.run(sql, params);
      },
      all: (...params: any[]) => {
        const stmt = db.prepare(sql);
        stmt.bind(params);
        const results: any[] = [];
        while (stmt.step()) {
          results.push(stmt.getAsObject());
        }
        stmt.free();
        return results;
      },
      get: (...params: any[]) => {
        const stmt = db.prepare(sql);
        stmt.bind(params);
        const result = stmt.step() ? stmt.getAsObject() : null;
        stmt.free();
        return result;
      }
    };
  }
  /**
   * Close database connection
   */
  public close(): void {
    if (this.saveInterval) {
      clearInterval(this.saveInterval);
      this.saveInterval = null;
    }
    if (this.db) {
      this.save(); // Final save
      this.db.close();
      this.db = null;
      logger.info('Database connection closed');
    }
  }
}
// Singleton instance
let dbManager: DatabaseManager | null = null;
let initPromise: Promise<DatabaseManager> | null = null;
export async function initDatabase(): Promise<DatabaseManager> {
  if (dbManager) return dbManager;
  if (!initPromise) {
    initPromise = (async () => {
      dbManager = new DatabaseManager();
      await dbManager.init();
      return dbManager;
    })();
  }
  return initPromise;
}
export function getDatabase(): DatabaseManager {
  if (!dbManager) {
    throw new Error('Database not initialized. Call initDatabase() first.');
  }
  return dbManager;
}
export function closeDatabase(): void {
  if (dbManager) {
    dbManager.close();
    dbManager = null;
    initPromise = null;
  }
}
