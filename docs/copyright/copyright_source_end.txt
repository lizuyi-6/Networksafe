import { useState, useEffect } from 'react';
import {
    Brain, Plus, Trash2, TestTube,
    Server, Laptop, Globe, Check, X, Loader2,
    ChevronDown, ChevronUp, AlertCircle
} from 'lucide-react';

interface AIEndpoint {
    id: string;
    name: string;
    url: string;
    modelName: string;
    apiKey: string;
    enabled: boolean;
    priority: number;
    timeout: number;
    type: 'ollama' | 'openai' | 'custom';
    lastTest?: {
        success: boolean;
        latency: number;
        timestamp: string;
    };
}

interface AddEndpointFormData {
    name: string;
    url: string;
    modelName: string;
    apiKey: string;
    type: 'ollama' | 'openai' | 'custom';
    priority: number;
}

const AIEndpointSettings = () => {
    const [endpoints, setEndpoints] = useState<AIEndpoint[]>([]);
    const [loading, setLoading] = useState(true);
    const [showAddForm, setShowAddForm] = useState(false);
    const [testingId, setTestingId] = useState<string | null>(null);

    const [formData, setFormData] = useState<AddEndpointFormData>({
        name: '',
        url: '',
        modelName: '',
        apiKey: '',
        type: 'ollama',
        priority: 10,
    });

    // 妯℃嫙绔偣鏁版嵁
    useEffect(() => {
        const mockEndpoints: AIEndpoint[] = [
            {
                id: 'default-laptop',
                name: '鏈湴绗旇鏈?(Ollama)',
                url: 'http://192.168.4.1:5000/api/analyze',
                modelName: 'llama3',
                apiKey: '',
                enabled: true,
                priority: 100,
                timeout: 10000,
                type: 'ollama',
                lastTest: {
                    success: true,
                    latency: 234,
                    timestamp: new Date().toISOString(),
                },
            },
        ];

        setTimeout(() => {
            setEndpoints(mockEndpoints);
            setLoading(false);
        }, 500);
    }, []);

    const handleAddEndpoint = () => {
        if (!formData.url) return;

        const newEndpoint: AIEndpoint = {
            id: `custom-${Date.now()}`,
            name: formData.name || `鑷畾涔夌鐐?${endpoints.length}`,
            url: formData.url,
            modelName: formData.modelName || 'default',
            apiKey: formData.apiKey,
            enabled: true,
            priority: formData.priority,
            timeout: 15000,
            type: formData.type,
        };

        setEndpoints(prev => [...prev, newEndpoint]);
        setShowAddForm(false);
        setFormData({
            name: '',
            url: '',
            modelName: '',
            apiKey: '',
            type: 'ollama',
            priority: 10,
        });
    };

    const handleDeleteEndpoint = (id: string) => {
        if (id === 'default-laptop') return;
        setEndpoints(prev => prev.filter(ep => ep.id !== id));
    };

    const handleToggleEndpoint = (id: string) => {
        setEndpoints(prev => prev.map(ep =>
            ep.id === id ? { ...ep, enabled: !ep.enabled } : ep
        ));
    };

    const handleTestEndpoint = async (id: string) => {
        setTestingId(id);

        // 妯℃嫙娴嬭瘯
        await new Promise(resolve => setTimeout(resolve, 1500));

        const success = Math.random() > 0.3;
        const latency = Math.floor(Math.random() * 500) + 100;

        setEndpoints(prev => prev.map(ep =>
            ep.id === id ? {
                ...ep,
                lastTest: {
                    success,
                    latency,
                    timestamp: new Date().toISOString(),
                },
            } : ep
        ));

        setTestingId(null);
    };

    const handlePriorityChange = (id: string, direction: 'up' | 'down') => {
        setEndpoints(prev => {
            const sorted = [...prev].sort((a, b) => a.priority - b.priority);
            const idx = sorted.findIndex(ep => ep.id === id);

            if (direction === 'up' && idx > 0) {
                const temp = sorted[idx].priority;
                sorted[idx].priority = sorted[idx - 1].priority;
                sorted[idx - 1].priority = temp;
            } else if (direction === 'down' && idx < sorted.length - 1) {
                const temp = sorted[idx].priority;
                sorted[idx].priority = sorted[idx + 1].priority;
                sorted[idx + 1].priority = temp;
            }

            return sorted;
        });
    };

    const getTypeIcon = (type: string) => {
        switch (type) {
            case 'ollama': return <Laptop size={16} className="text-emerald-500" />;
            case 'openai': return <Globe size={16} className="text-blue-500" />;
            default: return <Server size={16} className="text-orange-500" />;
        }
    };

    const sortedEndpoints = [...endpoints].sort((a, b) => a.priority - b.priority);

    return (
        <div className="bg-card border border-border rounded-xl p-5 shadow-sm">
            {/* Header */}
            <div className="flex justify-between items-center mb-5">
                <div className="flex items-center gap-2">
                    <Brain size={18} className="text-primary" />
                    <h3 className="font-semibold text-foreground">AI 妯″瀷绔偣閰嶇疆</h3>
                </div>
                <button
                    onClick={() => setShowAddForm(!showAddForm)}
                    className="flex items-center gap-1.5 px-3 py-1.5 text-xs font-medium bg-primary text-primary-foreground rounded-lg hover:bg-primary/90 transition-all"
                >
                    <Plus size={14} /> 娣诲姞绔偣
                </button>
            </div>

            {/* Info Banner */}
            <div className="bg-blue-500/10 border border-blue-500/20 rounded-lg p-3 mb-4 flex items-start gap-2">
                <AlertCircle size={16} className="text-blue-500 mt-0.5" />
                <div className="text-xs text-blue-700 dark:text-blue-300">
                    <strong>绂荤嚎妯″紡 AI 璋冪敤浼樺厛绾?</strong> 鎸夊垪琛ㄩ『搴忓皾璇曡皟鐢紝澶辫触鍒欒嚜鍔ㄩ檷绾у埌涓嬩竴涓鐐广€?
                    寤鸿灏嗗欢杩熸渶浣庣殑鏈嶅姟鍣ㄦ斁鍦ㄦ渶鍓嶉潰銆?
                </div>
            </div>

            {/* Add Form */}
            {showAddForm && (
                <div className="bg-muted/30 border border-border rounded-lg p-4 mb-4">
                    <div className="grid grid-cols-2 gap-4 mb-4">
                        <div>
                            <label className="block text-xs text-muted-foreground mb-1">绔偣鍚嶇О</label>
                            <input
                                type="text"
                                value={formData.name}
                                onChange={e => setFormData(prev => ({ ...prev, name: e.target.value }))}
                                placeholder="渚嬪: 鐢熶骇鏈嶅姟鍣ㄦā鍨?
                                className="w-full px-3 py-2 text-sm bg-background border border-border rounded-lg focus:border-primary focus:outline-none"
                            />
                        </div>
                        <div>
                            <label className="block text-xs text-muted-foreground mb-1">绫诲瀷</label>
                            <select
                                value={formData.type}
                                onChange={e => setFormData(prev => ({ ...prev, type: e.target.value as any }))}
                                className="w-full px-3 py-2 text-sm bg-background border border-border rounded-lg focus:border-primary focus:outline-none"
                            >
                                <option value="ollama">Ollama</option>
                                <option value="openai">OpenAI 鍏煎</option>
                                <option value="custom">鑷畾涔?/option>
                            </select>
                        </div>
                    </div>

                    <div className="mb-4">
                        <label className="block text-xs text-muted-foreground mb-1">API 绔偣 URL *</label>
                        <input
                            type="text"
                            value={formData.url}
                            onChange={e => setFormData(prev => ({ ...prev, url: e.target.value }))}
                            placeholder="http://your-server:5000/api/analyze"
                            className="w-full px-3 py-2 text-sm bg-background border border-border rounded-lg focus:border-primary focus:outline-none"
                        />
                    </div>

                    <div className="grid grid-cols-2 gap-4 mb-4">
                        <div>
                            <label className="block text-xs text-muted-foreground mb-1">妯″瀷鍚嶇О</label>
                            <input
                                type="text"
                                value={formData.modelName}
                                onChange={e => setFormData(prev => ({ ...prev, modelName: e.target.value }))}
                                placeholder="llama3 / gpt-4 / qwen"
                                className="w-full px-3 py-2 text-sm bg-background border border-border rounded-lg focus:border-primary focus:outline-none"
                            />
                        </div>

                            onClick={() => setShowAddForm(false)}
                            className="px-4 py-2 text-sm text-muted-foreground hover:text-foreground transition-colors"
                        >
                            鍙栨秷
                        </button>
                        <button
                            onClick={handleAddEndpoint}
                            disabled={!formData.url}
                            className="px-4 py-2 text-sm font-medium bg-primary text-primary-foreground rounded-lg hover:bg-primary/90 disabled:opacity-50 disabled:cursor-not-allowed transition-all"
                        >
                            娣诲姞
                        </button>
                    </div>
                </div>
            )}

            {/* Endpoints List */}
            {loading ? (
                <div className="text-center py-8 text-muted-foreground">
                    <Loader2 size={24} className="mx-auto mb-2 animate-spin" />
                    鍔犺浇涓?..
                </div>
            ) : sortedEndpoints.length === 0 ? (
                <div className="text-center py-8 text-muted-foreground">
                    鏆傛棤閰嶇疆鐨?AI 绔偣

            ) : (
                <div className="space-y-3">
                    {sortedEndpoints.map((endpoint, index) => (

                                    <div className="flex flex-col items-center">

                                            onClick={() => handlePriorityChange(endpoint.id, 'up')}
                                            disabled={index === 0}
                                            className="p-0.5 text-muted-foreground hover:text-foreground disabled:opacity-30"

                                            <ChevronUp size={14} />
                                        </button>
                                        <span className="text-xs font-bold text-muted-foreground">{index + 1}</span>

                                            onClick={() => handlePriorityChange(endpoint.id, 'down')}
                                            disabled={index === sortedEndpoints.length - 1}
                                            className="p-0.5 text-muted-foreground hover:text-foreground disabled:opacity-30"
                                        >
                                            <ChevronDown size={14} />
                                        </button>
                                    </div>

                                    <div>
                                        <div className="flex items-center gap-2 mb-1">
                                            {getTypeIcon(endpoint.type)}
                                            <span className="font-medium text-foreground">{endpoint.name}</span>
                                            {endpoint.id === 'default-laptop' && (
                                                <span className="px-1.5 py-0.5 text-xxs bg-primary/10 text-primary rounded">榛樿</span>
                                            )}
                                        </div>
                                        <div className="text-xs text-muted-foreground mb-1">{endpoint.url}</div>
                                        <div className="flex items-center gap-3 text-xs text-muted-foreground">

                                                <span className={`flex items-center gap-1 ${endpoint.lastTest.success ? 'text-emerald-500' : 'text-red-500'
                                                    }`}>
                                                    {endpoint.lastTest.success ? <Check size={12} /> : <X size={12} />}

                                    <button

                                        className="p-2 text-muted-foreground hover:text-foreground hover:bg-muted rounded-lg transition-all"
                                        title="娴嬭瘯杩炴帴"
                                    >
                                        {testingId === endpoint.id ? (
                                            <Loader2 size={16} className="animate-spin" />
                                        ) : (
                                            <TestTube size={16} />
                                        )}
                                    </button>
                                    <button
                                        onClick={() => handleToggleEndpoint(endpoint.id)}
                                        className={`p-2 rounded-lg transition-all ${endpoint.enabled
                                            ? 'text-emerald-500 hover:bg-emerald-500/10'
                                            : 'text-muted-foreground hover:bg-muted'
                                            }`}
                                        title={endpoint.enabled ? '绂佺敤' : '鍚敤'}
                                    >
                                        {endpoint.enabled ? <Check size={16} /> : <X size={16} />}
                                    </button>
                                    {endpoint.id !== 'default-laptop' && (
                                        <button
                                            onClick={() => handleDeleteEndpoint(endpoint.id)}
                                            className="p-2 text-red-500 hover:bg-red-500/10 rounded-lg transition-all"
                                            title="鍒犻櫎"
                                        >
                                            <Trash2 size={16} />
                                        </button>
                                    )}
                                </div>
                            </div>
                        </div>
                    ))}
                </div>
            )}

            {/* Footer */}
            <div className="mt-4 pt-4 border-t border-border text-xs text-muted-foreground">

                濡傛灉鎵€鏈夎嚜瀹氫箟绔偣澶辫触锛屽皢鍥為€€鍒版湰鍦扮瑪璁版湰妯″瀷銆?
            </div>

// Types for the subscription tiers
export type PlanTier = 'starter' | 'professional' | 'enterprise' | null;

interface LogoProps {
    tier?: PlanTier;
    className?: string;
    showText?: boolean;
}

const Logo: React.FC<LogoProps> = ({ tier, className = "", showText = true }) => {
    // Badge Configuration
    const getBadgeConfig = (tier: PlanTier) => {
        switch (tier) {
            case 'starter':
                return { text: 'STARTER', color: 'bg-gray-700 text-gray-200 border-gray-500' };
            case 'professional':
                return { text: 'PRO', color: 'bg-gradient-to-r from-blue-600 to-blue-400 text-white border-blue-400' };
            case 'enterprise':
                return { text: 'ULTRA', color: 'bg-gradient-to-r from-purple-600 to-red-500 text-white border-purple-400' };
            default:
                return null;
        }
    };

    const badge = getBadgeConfig(tier || null);

    return (
        <div className={`flex items-center gap-3 select-none ${className}`}>
            {/* Logo Image - Refined: No box, just the shield */}
            <div className="relative group w-10 h-10 transition-transform group-hover:scale-110 duration-300">
                <img
                    src={logoImg}
                    alt="FluxGuard Logo"
                    className="w-full h-full object-contain filter drop-shadow-[0_0_8px_rgba(59,130,246,0.5)]"
                />

                {/* Glow Effect based on tier */}
                {tier === 'professional' && <div className="absolute inset-0 bg-blue-500/40 blur-xl rounded-full -z-10 animate-pulse"></div>}
                {tier === 'enterprise' && <div className="absolute inset-0 bg-purple-500/40 blur-xl rounded-full -z-10 animate-pulse"></div>}
            </div>

            {/* Text & Badge */}
            {(showText || badge) && (
                <div className="flex flex-col justify-center">
                    <div className="flex items-center gap-2">
                        {showText && (
                            <span className="font-bold text-xl tracking-tight text-white group-hover:text-green-50 transition-colors">
                                FluxGuard
                            </span>
                        )}

                        {/* Dynamic Badge */}
                        {badge && (
                            <motion.div
                                initial={{ opacity: 0, scale: 0.8, x: -10 }}
                                animate={{ opacity: 1, scale: 1, x: 0 }}
                                key={tier} // Re-animate on toggle
                                className={`px-2 py-0.5 rounded-full text-[10px] font-bold border shadow-lg ${badge.color} backdrop-blur-md`}
                            >
                                {badge.text}
                            </motion.div>
                        )}
                    </div>
                </div>
            )}
        </div>
    );
};

import { createContext, useState, useContext, ReactNode } from 'react';

interface DemoContextType {
    isDemoMode: boolean;
    toggleDemoMode: () => void;
}

const DemoContext = createContext<DemoContextType | undefined>(undefined);

export const DemoProvider = ({ children }: { children: ReactNode }) => {
    const [isDemoMode, setIsDemoMode] = useState<boolean>(() => {
        return localStorage.getItem('demo_mode') === 'true';
    });

    const toggleDemoMode = () => {
        setIsDemoMode(prev => {
            const newValue = !prev;
            localStorage.setItem('demo_mode', String(newValue));
            window.location.reload(); // Reload to ensure all services/components pick up the change
            return newValue;
        });
    };

    return (
        <DemoContext.Provider value={{ isDemoMode, toggleDemoMode }}>
            {children}
        </DemoContext.Provider>
    );
};

export const useDemo = () => {
    const context = useContext(DemoContext);
    if (!context) {
        throw new Error('useDemo must be used within a DemoProvider');
    }
    return context;
};

const LanguageContext = createContext<LanguageContextType | undefined>(undefined);

export const LanguageProvider = ({ children }: { children: ReactNode }) => {
    const [language, setLanguage] = useState<Language>('en');

    const toggleLanguage = () => {
        setLanguage(prev => prev === 'en' ? 'zh' : 'en');
    };

    const t = (key: string): string => {
        const translation = translations[key];
        if (!translation) return key;
        return translation[language];
    };

    return (
        <LanguageContext.Provider value={{ language, toggleLanguage, t }}>
            {children}
        </LanguageContext.Provider>
    );
};

export const useLanguage = () => {
    const context = useContext(LanguageContext);
    if (!context) {
        throw new Error('useLanguage must be used within a LanguageProvider');
    }
    return context;
};
import React, { createContext, useContext, useEffect, useState } from 'react';

type Theme = 'dark' | 'light' | 'system';

type ThemeProviderState = {
    theme: Theme;
    setTheme: (theme: Theme) => void;
};

const initialState: ThemeProviderState = {
    theme: 'system',
    setTheme: () => null,
};

const ThemeContext = createContext<ThemeProviderState>(initialState);

export function ThemeProvider({
    children,
    defaultTheme = 'system',
    storageKey = 'vite-ui-theme',
}: {
    children: React.ReactNode;
    defaultTheme?: Theme;
    storageKey?: string;
}) {
    const [theme, setTheme] = useState<Theme>(
        () => (localStorage.getItem(storageKey) as Theme) || defaultTheme
    );

    useEffect(() => {
        const root = window.document.documentElement;

        root.classList.remove('light', 'dark');

        if (theme === 'system') {
            const systemTheme = window.matchMedia('(prefers-color-scheme: dark)')
                .matches
                ? 'dark'
                : 'light';

            root.classList.add(systemTheme);
            return;
        }

        root.classList.add(theme);
    }, [theme]);

    const value = {
        theme,
        setTheme: (theme: Theme) => {
            localStorage.setItem(storageKey, theme);
            setTheme(theme);
        },
    };

    return (
        <ThemeContext.Provider value={value}>
            {children}
        </ThemeContext.Provider>
    );
}

export const useTheme = () => {
    const context = useContext(ThemeContext);

    if (context === undefined)
        throw new Error('useTheme must be used within a ThemeProvider');

    return context;
};

            case 'performance': return <AlertTriangle size={18} />;
            default: return <Bell size={18} />;
        }
    };

    const filteredAlerts = filter === 'all' ? alerts : alerts.filter(a => a.severity.toLowerCase() === filter);

        setScanComplete(false);
        setProgress(0);
        setLogs([]);
        setVulnerabilities([]);

            { msg: 'Initializing scanner engine...', duration: 500, type: 'info' as const },
            { msg: 'Checking SSL/TLS configuration...', duration: 800, type: 'info' as const },
            { msg: 'Analyzing HTTP headers...', duration: 600, type: 'info' as const },
            { msg: 'Scanning for XSS vulnerabilities...', duration: 1000, type: 'info' as const },
            { msg: 'Checking SQL injection vectors...', duration: 900, type: 'info' as const },

            { msg: 'Checking rate limiting...', duration: 500, type: 'success' as const },
            { msg: 'Finalizing report...', duration: 400, type: 'success' as const },
        ];

        const runStep = () => {
            if (currentStep >= steps.length) {
                setIsScanning(false);
                setScanComplete(true);
                setProgress(100);
                addLog('Scan complete!', 'success');

                // Use API to get results (mock or real)
                fetchScannerResults().then(results => {
                    setVulnerabilities(results as any);
                });
                return;
            }

            const step = steps[currentStep];
            addLog(step.msg, step.type);
            setProgress(((currentStep + 1) / steps.length) * 100);
            currentStep++;

            setTimeout(runStep, step.duration);
        };

        setTimeout(runStep, 300);

        <div className="space-y-6 p-6">
            <div className="flex justify-between items-center">
                <div>

/**
 * 鍓嶇鍔犲瘑宸ュ叿 - AES-256-GCM
 * 绗﹀悎 OWASP 鍔犲瘑鎸囧崡鍜?Web Crypto API 鏈€浣冲疄璺?
 * 
 * 娉ㄦ剰: 
 * 1. 姝ゅ疄鐜颁娇鐢?Web Crypto API锛岄渶瑕?HTTPS 鐜
 * 2. 鐢熶骇鐜瀵嗛挜搴斾粠鏈嶅姟鍣ㄥ畨鍏ㄨ幏鍙栵紝涓嶅簲纭紪鐮?
 */

const ALGORITHM = 'AES-GCM';
const KEY_LENGTH = 256;
const IV_LENGTH = 12;
const PBKDF2_ITERATIONS = 100000;  // OWASP 2023 寤鸿鏇撮珮锛屾澶勪负鎬ц兘鎶樹腑

/**
 * 灏嗗瓧绗︿覆杞崲涓?ArrayBuffer
 */
function stringToArrayBuffer(str: string): ArrayBuffer {
    const encoder = new TextEncoder();
    return encoder.encode(str).buffer;
}

/**
 * 灏?ArrayBuffer 杞崲涓?Base64 瀛楃涓?
 */
function arrayBufferToBase64(buffer: any): string {
    const bytes = new Uint8Array(buffer);
    let binary = '';
    for (let i = 0; i < bytes.byteLength; i++) {
        binary += String.fromCharCode(bytes[i]);
    }
    return btoa(binary);
}

/**
 * 灏?Base64 瀛楃涓茶浆鎹负 ArrayBuffer
 */
function base64ToArrayBuffer(base64: string): ArrayBuffer {
    const binary = atob(base64);
    const bytes = new Uint8Array(binary.length);
    for (let i = 0; i < binary.length; i++) {
        bytes[i] = binary.charCodeAt(i);
    }
    return bytes.buffer;
}



/**
 * 浣跨敤 PBKDF2 浠庡瘑鐮佹淳鐢熷瘑閽?
 * @param password 鐢ㄦ埛瀵嗙爜
 * @param salt 鐩愬€?(16 瀛楄妭)
 * @returns 娲剧敓鐨?CryptoKey
 */
export async function deriveKeyFromPassword(
    password: string,
    salt: Uint8Array
): Promise<CryptoKey> {
    const passwordBuffer = stringToArrayBuffer(password);

    const baseKey = await crypto.subtle.importKey(
        'raw',
        passwordBuffer,
        'PBKDF2',
        false,
        ['deriveKey']
    );

    return crypto.subtle.deriveKey(
        {
            name: 'PBKDF2',
            salt: salt as any,
            iterations: PBKDF2_ITERATIONS,
            hash: 'SHA-256',
        },
        baseKey,
        { name: ALGORITHM, length: KEY_LENGTH },
        false,
        ['encrypt', 'decrypt']
    );
}

/**
 * 鐢熸垚闅忔満鐩愬€?
 */
export function generateSalt(): Uint8Array {
    return crypto.getRandomValues(new Uint8Array(16));
}

/**
 * 浣跨敤瀵嗛挜鍔犲瘑鏁版嵁
 * @param plaintext 鏄庢枃
 * @param key CryptoKey 瀵硅薄
 * @returns Base64 缂栫爜鐨勫姞瀵嗘暟鎹?(鏍煎紡: iv:ciphertext)
 */
export async function encryptWithKey(plaintext: string, key: CryptoKey): Promise<string> {
    const iv = crypto.getRandomValues(new Uint8Array(IV_LENGTH));
    const data = stringToArrayBuffer(plaintext);

    const encrypted = await crypto.subtle.encrypt(
        { name: ALGORITHM, iv },
        key,
        data
    );

    const ivBase64 = arrayBufferToBase64(iv.buffer);
    const ciphertextBase64 = arrayBufferToBase64(encrypted);

    return `${ivBase64}:${ciphertextBase64}`;
}

/**
 * 浣跨敤瀵嗛挜瑙ｅ瘑鏁版嵁
 * @param encryptedData Base64 缂栫爜鐨勫姞瀵嗘暟鎹?
 * @param key CryptoKey 瀵硅薄
 * @returns 瑙ｅ瘑鍚庣殑鏄庢枃
 */
export async function decryptWithKey(encryptedData: string, key: CryptoKey): Promise<string> {
    const [ivBase64, ciphertextBase64] = encryptedData.split(':');

    if (!ivBase64 || !ciphertextBase64) {
        throw new Error('鏃犳晥鐨勫姞瀵嗘暟鎹牸寮?);
    }

    const iv = new Uint8Array(base64ToArrayBuffer(ivBase64));

    if (iv.length !== IV_LENGTH) {
        throw new Error(`鏃犳晥鐨?IV 闀垮害: 鏈熸湜 ${IV_LENGTH}锛屽疄闄?${iv.length}`);
    }

    const ciphertext = base64ToArrayBuffer(ciphertextBase64);

    const decrypted = await crypto.subtle.decrypt(
        { name: ALGORITHM, iv },
        key,
        ciphertext as any
    );

    const decoder = new TextDecoder();
    return decoder.decode(decrypted);
}

/**
 * 浣跨敤瀵嗙爜鍔犲瘑 (鑷姩娲剧敓瀵嗛挜)
 * @param plaintext 鏄庢枃
 * @param password 瀵嗙爜
 * @returns Base64 缂栫爜鐨勫姞瀵嗘暟鎹?(鏍煎紡: salt:iv:ciphertext)
 */
export async function encryptWithPassword(plaintext: string, password: string): Promise<string> {
    const salt = generateSalt();
    const key = await deriveKeyFromPassword(password, salt);
    const encrypted = await encryptWithKey(plaintext, key);

    const saltBase64 = arrayBufferToBase64(salt.buffer);
    return `${saltBase64}:${encrypted}`;
}

/**
 * 浣跨敤瀵嗙爜瑙ｅ瘑
 * @param encryptedData 鍔犲瘑鏁版嵁 (鏍煎紡: salt:iv:ciphertext)
 * @param password 瀵嗙爜
 * @returns 瑙ｅ瘑鍚庣殑鏄庢枃
 */
export async function decryptWithPassword(encryptedData: string, password: string): Promise<string> {
    const parts = encryptedData.split(':');
    if (parts.length !== 3) {
        throw new Error('鏃犳晥鐨勫姞瀵嗘暟鎹牸寮?);
    }

    const [saltBase64, ivBase64, ciphertextBase64] = parts;
    const salt = new Uint8Array(base64ToArrayBuffer(saltBase64));
    const key = await deriveKeyFromPassword(password, salt);

    return decryptWithKey(`${ivBase64}:${ciphertextBase64}`, key);
}

/**
 * 妫€鏌ユ祻瑙堝櫒鏄惁鏀寔 Web Crypto API
 */
export function isCryptoSupported(): boolean {
    return typeof crypto !== 'undefined' && typeof crypto.subtle !== 'undefined';
}

/**
 * 鍔犲瘑鏁忔劅閰嶇疆瀵硅薄涓殑鎸囧畾瀛楁
 * @param obj 閰嶇疆瀵硅薄
 * @param sensitiveKeys 闇€瑕佸姞瀵嗙殑瀛楁鍚?
 * @param password 鍔犲瘑瀵嗙爜
 */
export async function encryptSensitiveFields<T extends Record<string, any>>(
    obj: T,
    sensitiveKeys: (keyof T)[],
    password: string
): Promise<T> {
    if (!isCryptoSupported()) {
        throw new Error('娴忚鍣ㄤ笉鏀寔 Web Crypto API锛岃浣跨敤 HTTPS');
    }

    const result = { ...obj };

    for (const key of sensitiveKeys) {
        if (result[key] && typeof result[key] === 'string') {
            result[key] = await encryptWithPassword(result[key] as string, password) as any;
        }
    }

    return result;
}
