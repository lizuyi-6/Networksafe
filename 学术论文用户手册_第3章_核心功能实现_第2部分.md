### 3.3 并发控制系统实现

#### 3.3.1 并发控制算法设计

并发控制系统采用令牌桶算法结合公平队列调度，实现精细化的资源管理：

```typescript
class ConcurrencyController {
  private buckets: Map<string, TokenBucket> = new Map();
  private queues: Map<string, RequestQueue> = new Map();
  private readonly defaultCapacity: number = 10;
  private readonly refillRate: number = 1; // 令牌/秒

  async acquireSlot(
    resourceId: string,
    requestId: string,
    priority: number = 1
  ): Promise<AcquisitionResult> {
    // 获取或创建令牌桶
    let bucket = this.buckets.get(resourceId);
    if (!bucket) {
      bucket = new TokenBucket(this.defaultCapacity, this.refillRate);
      this.buckets.set(resourceId, bucket);
    }

    // 获取或创建请求队列
    let queue = this.queues.get(resourceId);
    if (!queue) {
      queue = new RequestQueue();
      this.queues.set(resourceId, queue);
    }

    // 尝试获取令牌
    const tokenAcquired = await bucket.tryAcquire(1);

    if (tokenAcquired) {
      // 成功获取令牌，立即处理
      return {
        acquired: true,
        waitTime: 0,
        queuePosition: 0,
        estimatedProcessingTime: this.estimateProcessingTime(requestId),
        resourceId,
        requestId
      };
    } else {
      // 令牌不足，加入队列等待
      const queuePosition = await queue.enqueue(requestId, priority);

      return {
        acquired: false,
        waitTime: this.estimateWaitTime(queuePosition),
        queuePosition,
        estimatedProcessingTime: this.estimateProcessingTime(requestId),
        resourceId,
        requestId
      };
    }
  }

  async releaseSlot(resourceId: string, requestId: string): Promise<void> {
    // 从队列中移除请求（如果存在）
    const queue = this.queues.get(resourceId);
    if (queue) {
      await queue.dequeue(requestId);
    }

    // 通知等待的请求
    await this.notifyNextInQueue(resourceId);
  }

  private async notifyNextInQueue(resourceId: string): Promise<void> {
    const queue = this.queues.get(resourceId);
    const bucket = this.buckets.get(resourceId);

    if (!queue || !bucket) return;

    // 检查是否有可用令牌
    if (await bucket.hasAvailableTokens(1)) {
      // 从队列中获取下一个请求
      const nextRequest = await queue.peek();
      if (nextRequest) {
        // 触发请求处理
        await this.triggerRequestProcessing(resourceId, nextRequest.requestId);
      }
    }
  }

  private estimateWaitTime(queuePosition: number): number {
    // 基于队列位置和平均处理时间估算等待时间
    const averageProcessingTime = 100; // 毫秒
    return queuePosition * averageProcessingTime;
  }

  private estimateProcessingTime(requestId: string): number {
    // 基于请求特征估算处理时间
    // 这里可以基于历史数据或请求内容进行更精确的估算
    return 50 + Math.random() * 100; // 50-150毫秒
  }
}

class TokenBucket {
  private capacity: number;
  private tokens: number;
  private lastRefillTime: number;
  private refillRate: number; // 令牌/毫秒

  constructor(capacity: number, refillRatePerSecond: number) {
    this.capacity = capacity;
    this.tokens = capacity;
    this.lastRefillTime = Date.now();
    this.refillRate = refillRatePerSecond / 1000; // 转换为毫秒
  }

  async tryAcquire(tokens: number): Promise<boolean> {
    await this.refill();

    if (this.tokens >= tokens) {
      this.tokens -= tokens;
      return true;
    }

    return false;
  }

  async hasAvailableTokens(tokens: number): Promise<boolean> {
    await this.refill();
    return this.tokens >= tokens;
  }

  private async refill(): Promise<void> {
    const now = Date.now();
    const timePassed = now - this.lastRefillTime;

    // 计算应补充的令牌数
    const tokensToAdd = timePassed * this.refillRate;
    this.tokens = Math.min(this.capacity, this.tokens + tokensToAdd);
    this.lastRefillTime = now;
  }
}

class RequestQueue {
  private queue: PriorityQueue<QueueItem> = new PriorityQueue<QueueItem>(
    (a, b) => b.priority - a.priority // 优先级高的先出队
  );
  private requestMap: Map<string, QueueItem> = new Map();

  async enqueue(requestId: string, priority: number): Promise<number> {
    const item: QueueItem = {
      requestId,
      priority,
      enqueueTime: Date.now(),
      position: this.queue.size() + 1
    };

    this.queue.push(item);
    this.requestMap.set(requestId, item);

    // 更新所有项的位置
    this.updatePositions();

    return item.position;
  }

  async dequeue(requestId: string): Promise<boolean> {
    const item = this.requestMap.get(requestId);
    if (!item) return false;

    // 从优先队列中移除
    this.queue.remove(item);

    // 从映射中移除
    this.requestMap.delete(requestId);

    // 更新剩余项的位置
    this.updatePositions();

    return true;
  }

  async peek(): Promise<QueueItem | null> {
    if (this.queue.isEmpty()) return null;
    return this.queue.peek();
  }

  private updatePositions(): void {
    const items = this.queue.toArray();
    items.forEach((item, index) => {
      item.position = index + 1;
    });
  }
}
```

#### 3.3.2 资源隔离机制

系统实现细粒度的资源隔离，防止单个用户或请求占用过多资源：

```typescript
class ResourceIsolationManager {
  private isolationGroups: Map<string, IsolationGroup> = new Map();
  private resourceMonitors: Map<string, ResourceMonitor> = new Map();

  async initialize(): Promise<void> {
    // 初始化默认隔离组
    await this.initializeDefaultGroups();

    // 启动资源监控
    await this.startResourceMonitoring();
  }

  private async initializeDefaultGroups(): Promise<void> {
    const defaultGroups: IsolationGroup[] = [
      {
        id: 'default',
        name: '默认组',
        limits: {
          cpu: 0.5,      // 50% CPU使用率
          memory: 512,   // 512MB内存
          connections: 100, // 100个并发连接
          bandwidth: 1024 // 1MB/s带宽
        },
        priority: 1
      },
      {
        id: 'premium',
        name: '高级用户组',
        limits: {
          cpu: 0.8,
          memory: 1024,
          connections: 500,
          bandwidth: 5120
        },
        priority: 2
      },
      {
        id: 'suspicious',
        name: '可疑用户组',
        limits: {
          cpu: 0.2,
          memory: 256,
          connections: 10,
          bandwidth: 100
        },
        priority: 0
      }
    ];

    for (const group of defaultGroups) {
      this.isolationGroups.set(group.id, group);
    }
  }

  async checkResourceLimits(
    userId: string,
    resourceType: ResourceType,
    requestedAmount: number
  ): Promise<ResourceCheckResult> {
    const groupId = await this.getUserGroup(userId);
    const group = this.isolationGroups.get(groupId);
    if (!group) {
      return { allowed: false, reason: 'group_not_found' };
    }

    const monitor = await this.getResourceMonitor(userId, resourceType);
    const currentUsage = await monitor.getCurrentUsage();
    const limit = group.limits[resourceType];

    if (currentUsage + requestedAmount > limit) {
      return {
        allowed: false,
        reason: 'resource_limit_exceeded',
        currentUsage,
        limit,
        requestedAmount,
        available: Math.max(0, limit - currentUsage)
      };
    }

    return {
      allowed: true,
      currentUsage,
      limit,
      requestedAmount,
      available: limit - currentUsage - requestedAmount
    };
  }

  async allocateResource(
    userId: string,
    resourceType: ResourceType,
    amount: number
  ): Promise<AllocationResult> {
    const checkResult = await this.checkResourceLimits(userId, resourceType, amount);
    if (!checkResult.allowed) {
      return {
        allocated: false,
        reason: checkResult.reason,
        details: checkResult
      };
    }

    const monitor = await this.getResourceMonitor(userId, resourceType);
    await monitor.allocate(amount);

    return {
      allocated: true,
      allocationId: this.generateAllocationId(userId, resourceType),
      amount,
      expiresAt: new Date(Date.now() + 5 * 60 * 1000) // 5分钟后自动释放
    };
  }

  async releaseResource(
    userId: string,
    resourceType: ResourceType,
    allocationId: string
  ): Promise<void> {
    const monitor = await this.getResourceMonitor(userId, resourceType);
    await monitor.release(allocationId);
  }

  private async getUserGroup(userId: string): Promise<string> {
    // 这里可以根据用户行为、订阅等级等因素动态分配组
    // 简化实现：根据用户ID哈希分配
    const hash = this.hashString(userId);
    const groups = Array.from(this.isolationGroups.keys());
    const groupIndex = hash % groups.length;
    return groups[groupIndex];
  }

  private async getResourceMonitor(
    userId: string,
    resourceType: ResourceType
  ): Promise<ResourceMonitor> {
    const key = `${userId}:${resourceType}`;
    let monitor = this.resourceMonitors.get(key);

    if (!monitor) {
      monitor = new ResourceMonitor(userId, resourceType);
      this.resourceMonitors.set(key, monitor);
    }

    return monitor;
  }
}

class ResourceMonitor {
  private currentUsage: number = 0;
  private allocations: Map<string, Allocation> = new Map();
  private readonly cleanupInterval: number = 60000; // 1分钟清理一次

  constructor(
    private userId: string,
    private resourceType: ResourceType
  ) {
    this.startCleanupTimer();
  }

  async getCurrentUsage(): Promise<number> {
    await this.cleanupExpiredAllocations();
    return this.currentUsage;
  }

  async allocate(amount: number): Promise<string> {
    const allocationId = this.generateAllocationId();
    const allocation: Allocation = {
      id: allocationId,
      amount,
      allocatedAt: Date.now(),
      expiresAt: Date.now() + 5 * 60 * 1000 // 5分钟过期
    };

    this.allocations.set(allocationId, allocation);
    this.currentUsage += amount;

    return allocationId;
  }

  async release(allocationId: string): Promise<boolean> {
    const allocation = this.allocations.get(allocationId);
    if (!allocation) return false;

    this.currentUsage -= allocation.amount;
    this.allocations.delete(allocationId);

    return true;
  }

  private async cleanupExpiredAllocations(): Promise<void> {
    const now = Date.now();
    const expiredAllocations: string[] = [];

    for (const [allocationId, allocation] of this.allocations) {
      if (allocation.expiresAt <= now) {
        expiredAllocations.push(allocationId);
      }
    }

    for (const allocationId of expiredAllocations) {
      await this.release(allocationId);
    }
  }

  private startCleanupTimer(): void {
    setInterval(() => {
      this.cleanupExpiredAllocations();
    }, this.cleanupInterval);
  }
}
```

```typescript
class MultiFactorAuthentication {
  private factors: Map<string, AuthenticationFactor> = new Map();
  private readonly requiredFactors: Map<string, string[]> = new Map();

  async initialize(): Promise<void> {
    // 注册支持的认证因素
    await this.registerFactors();

    // 配置默认的认证要求
    await this.configureDefaultRequirements();
  }

  private async registerFactors(): Promise<void> {
    const factors: AuthenticationFactor[] = [
      {
        id: 'password',
        name: '密码认证',
        type: 'knowledge',
        implementation: new PasswordAuthenticator(),
        strength: 'medium',
        cost: 'low'
      },
      {
        id: 'totp',
        name: '时间型一次性密码',
        type: 'possession',
        implementation: new TOTPAuthenticator(),
        strength: 'high',
        cost: 'medium'
      },
      {
        id: 'sms',
        name: '短信验证码',
        type: 'possession',
        implementation: new SMSAuthenticator(),
        strength: 'medium',
        cost: 'medium'
      },
      {
        id: 'biometric',
        name: '生物特征认证',
        type: 'inherence',
        implementation: new BiometricAuthenticator(),
        strength: 'high',
        cost: 'high'
      },
      {
        id: 'security_key',
        name: '安全密钥',
        type: 'possession',
        implementation: new SecurityKeyAuthenticator(),
        strength: 'very_high',
        cost: 'high'
      }
    ];

    for (const factor of factors) {
      this.factors.set(factor.id, factor);
    }
  }

  private async configureDefaultRequirements(): Promise<void> {
    // 配置不同安全级别的认证要求
    this.requiredFactors.set('low', ['password']);
    this.requiredFactors.set('medium', ['password', 'sms']);
    this.requiredFactors.set('high', ['password', 'totp']);
    this.requiredFactors.set('very_high', ['password', 'security_key', 'biometric']);
  }

  async authenticate(
    userId: string,
    securityLevel: SecurityLevel = 'medium',
    context: AuthenticationContext
  ): Promise<AuthenticationResult> {
    const requiredFactorIds = this.requiredFactors.get(securityLevel) || [];
    const completedFactors: CompletedFactor[] = [];
    const failedFactors: FailedFactor[] = [];

    // 检查每个必需的认证因素
    for (const factorId of requiredFactorIds) {
      const factor = this.factors.get(factorId);
      if (!factor) {
        failedFactors.push({
          factorId,
          reason: 'factor_not_found',
          timestamp: new Date()
        });
        continue;
      }

      try {
        const factorResult = await factor.implementation.authenticate(userId, context);

        if (factorResult.success) {
          completedFactors.push({
            factorId,
            factorName: factor.name,
            timestamp: new Date(),
            metadata: factorResult.metadata
          });
        } else {
          failedFactors.push({
            factorId,
            reason: factorResult.reason,
            details: factorResult.details,
            timestamp: new Date()
          });
        }
      } catch (error) {
        failedFactors.push({
          factorId,
          reason: 'authentication_error',
          details: error.message,
          timestamp: new Date()
        });
      }
    }

    // 评估认证结果
    const success = failedFactors.length === 0;
    const score = this.calculateAuthenticationScore(completedFactors, failedFactors);

    return {
      success,
      score,
      completedFactors,
      failedFactors,
      securityLevel,
      timestamp: new Date(),
      sessionToken: success ? await this.generateSessionToken(userId, completedFactors) : null
    };
  }

  private calculateAuthenticationScore(
    completed: CompletedFactor[],
    failed: FailedFactor[]
  ): number {
    if (failed.length > 0) return 0;

    let totalScore = 0;
    let maxScore = 0;

    for (const factor of completed) {
      const factorConfig = this.factors.get(factor.factorId);
      if (!factorConfig) continue;

      // 根据因素强度计算分数
      const factorScore = this.getFactorScore(factorConfig.strength);
      totalScore += factorScore;
      maxScore += factorScore;
    }

    return maxScore > 0 ? totalScore / maxScore : 0;
  }

  private getFactorScore(strength: FactorStrength): number {
    switch (strength) {
      case 'very_high': return 1.0;
      case 'high': return 0.8;
      case 'medium': return 0.6;
      case 'low': return 0.4;
      default: return 0.5;
    }
  }

  private async generateSessionToken(
    userId: string,
    factors: CompletedFactor[]
  ): Promise<string> {
    // 生成包含认证信息的JWT令牌
    const payload = {
      sub: userId,
      auth_factors: factors.map(f => f.factorId),
      auth_score: this.calculateAuthenticationScore(factors, []),
      iat: Math.floor(Date.now() / 1000),
      exp: Math.floor(Date.now() / 1000) + 3600 // 1小时过期
    };

    const secret = process.env.JWT_SECRET || 'default_secret';
    return jwt.sign(payload, secret, { algorithm: 'HS256' });
  }
}

class PasswordAuthenticator {
  async authenticate(
    userId: string,
    context: AuthenticationContext
  ): Promise<FactorAuthenticationResult> {
    const { password } = context.credentials;

    if (!password) {
      return {
        success: false,
        reason: 'password_required',
        timestamp: new Date()
      };
    }

    // 从数据库获取用户密码哈希
    const user = await this.getUser(userId);
    if (!user) {
      return {
        success: false,
        reason: 'user_not_found',
        timestamp: new Date()
      };
    }

    // 验证密码
    const isValid = await bcrypt.compare(password, user.passwordHash);

    if (!isValid) {
      // 记录失败尝试
      await this.recordFailedAttempt(userId);

      return {
        success: false,
        reason: 'invalid_password',
        timestamp: new Date(),
        details: {
          attemptCount: await this.getFailedAttemptCount(userId),
          locked: await this.isAccountLocked(userId)
        }
      };
    }

    // 重置失败计数
    await this.resetFailedAttempts(userId);

    return {
      success: true,
      timestamp: new Date(),
      metadata: {
        lastPasswordChange: user.lastPasswordChange,
        passwordStrength: this.assessPasswordStrength(password)
      }
    };
  }

  private async recordFailedAttempt(userId: string): Promise<void> {
    const key = `auth:failed_attempts:${userId}`;
    const now = Date.now();

    // 使用Redis记录失败尝试
    await redisClient.zadd(key, now, now.toString());
    await redisClient.expire(key, 900); // 15分钟过期

    // 检查是否达到锁定阈值
    const attempts = await redisClient.zcount(key, now - 900000, now); // 15分钟内
    if (attempts >= 5) {
      await this.lockAccount(userId);
    }
  }

  private async lockAccount(userId: string): Promise<void> {
    const lockKey = `auth:locked:${userId}`;
    await redisClient.setex(lockKey, 900, 'locked'); // 锁定15分钟
  }

  private assessPasswordStrength(password: string): PasswordStrength {
    let score = 0;

    // 长度检查
    if (password.length >= 12) score += 2;
    else if (password.length >= 8) score += 1;

    // 字符多样性检查
    if (/[a-z]/.test(password)) score += 1;
    if (/[A-Z]/.test(password)) score += 1;
    if (/[0-9]/.test(password)) score += 1;
    if (/[^a-zA-Z0-9]/.test(password)) score += 1;

    // 常见密码检查
    const commonPasswords = ['password', '123456', 'qwerty'];
    if (commonPasswords.includes(password.toLowerCase())) {
      score = 0;
    }

    // 转换为强度等级
    if (score >= 5) return 'very_strong';
    if (score >= 4) return 'strong';
    if (score >= 3) return 'medium';
    return 'weak';
  }
}
```

#### 3.4.2 会话安全管理

系统实现安全的会话管理机制，防止会话劫持和固定攻击：

```typescript
class SessionManager {
  private readonly sessionTimeout: number = 3600000; // 1小时
  private readonly refreshInterval: number = 300000; // 5分钟
  private readonly maxSessionsPerUser: number = 5;

  async createSession(
    userId: string,
    deviceInfo: DeviceInfo,
    ipAddress: string
  ): Promise<Session> {
    // 检查会话数量限制
    const currentSessions = await this.getUserSessions(userId);
    if (currentSessions.length >= this.maxSessionsPerUser) {
      // 移除最旧的会话
      const oldestSession = currentSessions.sort((a, b) =>
        a.createdAt.getTime() - b.createdAt.getTime()
      )[0];
      await this.invalidateSession(oldestSession.id);
    }

    // 生成会话ID
    const sessionId = this.generateSessionId();

    // 创建会话令牌
    const sessionToken = await this.generateSessionToken(sessionId, userId);

    // 存储会话信息
    const session: Session = {
      id: sessionId,
      userId,
      token: sessionToken,
      deviceInfo,
      ipAddress,
      createdAt: new Date(),
      lastActivity: new Date(),
      expiresAt: new Date(Date.now() + this.sessionTimeout),
      isValid: true
    };

    await this.storeSession(session);

    // 启动会话刷新定时器
    this.startSessionRefresh(sessionId);

    return session;
  }

  async validateSession(sessionToken: string, ipAddress: string): Promise<ValidationResult> {
    try {
      // 解码会话令牌
      const decoded = await this.decodeSessionToken(sessionToken);
      if (!decoded) {
        return { valid: false, reason: 'invalid_token' };
      }

      const { sessionId, userId } = decoded;

      // 获取会话信息
      const session = await this.getSession(sessionId);
      if (!session) {
        return { valid: false, reason: 'session_not_found' };
      }

      // 检查会话状态
      if (!session.isValid) {
        return { valid: false, reason: 'session_invalidated' };
      }

      // 检查过期时间
      if (session.expiresAt < new Date()) {
        return { valid: false, reason: 'session_expired' };
      }

      // 检查IP地址变化（可选）
      if (session.ipAddress !== ipAddress) {
        // 记录IP变化，但不一定立即失效
        await this.recordIpChange(sessionId, ipAddress);

        // 如果IP变化频繁，可以触发重新认证
        const ipChanges = await this.getIpChangeCount(sessionId);
        if (ipChanges > 3) {
          return { valid: false, reason: 'suspicious_ip_changes' };
        }
      }

      // 更新最后活动时间
      await this.updateLastActivity(sessionId);

      return {
        valid: true,
        sessionId,
        userId,
        deviceInfo: session.deviceInfo,
        remainingTime: session.expiresAt.getTime() - Date.now()
      };
    } catch (error) {
      console.error('Session validation error:', error);
      return { valid: false, reason: 'validation_error' };
    }
  }

  async invalidateSession(sessionId: string): Promise<boolean> {
    try {
      const session = await this.getSession(sessionId);
      if (!session) return false;

      // 标记会话为无效
      session.isValid = false;
      session.invalidatedAt = new Date();

      await this.updateSession(session);

      // 停止刷新定时器
      this.stopSessionRefresh(sessionId);

      return true;
    } catch (error) {
      console.error('Session invalidation error:', error);
      return false;
    }
  }

  async invalidateUserSessions(userId: string, excludeSessionId?: string): Promise<number> {
    const sessions = await this.getUserSessions(userId);
    let invalidatedCount = 0;

    for (const session of sessions) {
      if (excludeSessionId && session.id === excludeSessionId) {
        continue;
      }

      if (await this.invalidateSession(session.id)) {
        invalidatedCount++;
      }
    }

    return invalidatedCount;
  }

  private async generateSessionToken(sessionId: string, userId: string): Promise<string> {
    const payload = {
      sessionId,
      userId,
      iat: Math.floor(Date.now() / 1000),
      exp: Math.floor(Date.now() / 1000) + this.sessionTimeout / 1000
    };

    const secret = this.getSessionSecret();
    return jwt.sign(payload, secret, {
      algorithm: 'HS256',
      header: {
        typ: 'JWT',
        alg: 'HS256'
      }
    });
  }

  private async decodeSessionToken(token: string): Promise<{ sessionId: string; userId: string } | null> {
    try {
      const secret = this.getSessionSecret();
      const decoded = jwt.verify(token, secret, { algorithms: ['HS256'] }) as any;

      return {
        sessionId: decoded.sessionId,
        userId: decoded.userId
      };
    } catch (error) {
      console.error('Token decode error:', error);
      return null;
    }
  }

  private startSessionRefresh(sessionId: string): void {
    const intervalId = setInterval(async () => {
      try {
        const session = await this.getSession(sessionId);
        if (!session || !session.isValid) {
          this.stopSessionRefresh(sessionId);
          return;
        }

        // 检查是否需要刷新
        const timeUntilExpiry = session.expiresAt.getTime() - Date.now();
        if (timeUntilExpiry < this.refreshInterval) {
          // 刷新会话
          session.expiresAt = new Date(Date.now() + this.sessionTimeout);
          await this.updateSession(session);
        }
      } catch (error) {
        console.error('Session refresh error:', error);
        this.stopSessionRefresh(sessionId);
      }
    }, this.refreshInterval);

    // 存储定时器ID以便后续停止
    this.sessionTimers.set(sessionId, intervalId);
  }

  private stopSessionRefresh(sessionId: string): void {
    const intervalId = this.sessionTimers.get(sessionId);
    if (intervalId) {
      clearInterval(intervalId);
      this.sessionTimers.delete(sessionId);
    }
  }
}
```
