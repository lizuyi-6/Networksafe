反机器人防护系统采用多维度检测算法，结合规则引擎和机器学习模型，实现高精度的机器人识别。检测算法架构如图3-1所示，包含以下核心组件：

```typescript
      timeOfDay: new Date().getHours(),
      dayOfWeek: new Date().getDay(),
      isWorkingHours: this.isWorkingHours()
    };

    return features;
  }

  // 具体特征提取方法实现
  private extractClientIP(request: Request): string {
    // 考虑X-Forwarded-For等代理头部
    const xForwardedFor = request.headers['x-forwarded-for'];
    if (xForwardedFor && typeof xForwardedFor === 'string') {
      return xForwardedFor.split(',')[0].trim();
    }
    return request.ip || request.connection.remoteAddress || 'unknown';
  }

  private async calculateRequestFrequency(request: Request): Promise<number> {
    const ip = this.extractClientIP(request);
    const key = `request_frequency:${ip}`;

    // 使用Redis有序集合计算请求频率
    const now = Date.now();
    const oneMinuteAgo = now - 60 * 1000;

    await redisClient.zadd(key, now, `${now}:${Math.random()}`);
    await redisClient.zremrangebyscore(key, 0, oneMinuteAgo);
    await redisClient.expire(key, 120);

    const count = await redisClient.zcount(key, oneMinuteAgo, now);
    return count;
  }
}
```

##### 3.1.1.2 规则引擎实现

规则引擎采用声明式规则定义，支持动态加载和实时生效：

```typescript
interface DetectionRule {
  id: string;
  name: string;
  description: string;
  severity: 'low' | 'medium' | 'high' | 'critical';
  condition: RuleCondition;
  action: RuleAction;
  enabled: boolean;
  createdAt: Date;
  updatedAt: Date;
}

type RuleCondition = {
  type: 'and' | 'or' | 'not' | 'comparison' | 'regex' | 'custom';
  children?: RuleCondition[];
  field?: string;
  operator?: 'eq' | 'ne' | 'gt' | 'lt' | 'gte' | 'lte' | 'contains' | 'matches';
  value?: any;
  customFunction?: string;
};

type RuleAction = {
  type: 'log' | 'alert' | 'block' | 'rate_limit' | 'challenge';
  parameters: Record<string, any>;
};

class RuleEngine {
  private rules: DetectionRule[] = [];
  private ruleCache: Map<string, Function> = new Map();

  async evaluate(request: Request, features: RequestFeatures): Promise<RuleEvaluationResult[]> {
    const results: RuleEvaluationResult[] = [];

    for (const rule of this.rules) {
      if (!rule.enabled) continue;

      try {
        const conditionResult = await this.evaluateCondition(rule.condition, features);

        if (conditionResult) {
          results.push({
            ruleId: rule.id,
            ruleName: rule.name,
            severity: rule.severity,
            matched: true,
            action: rule.action,
            timestamp: new Date(),
            features: this.extractRelevantFeatures(features, rule.condition)
          });
        }
      } catch (error) {
        console.error(`Error evaluating rule ${rule.id}:`, error);
      }
    }

    return results;
  }

  private async evaluateCondition(condition: RuleCondition, features: RequestFeatures): Promise<boolean> {
    switch (condition.type) {
      case 'and':
        if (!condition.children) return false;
        for (const child of condition.children) {
          if (!await this.evaluateCondition(child, features)) {
            return false;
          }
        }
        return true;

      case 'or':
        if (!condition.children) return false;
        for (const child of condition.children) {
          if (await this.evaluateCondition(child, features)) {
            return true;
          }
        }
        return false;

      case 'not':
        if (!condition.children || condition.children.length !== 1) return false;
        return !await this.evaluateCondition(condition.children[0], features);

      case 'comparison':
        return this.evaluateComparison(condition, features);

      case 'regex':
        return this.evaluateRegex(condition, features);

      case 'custom':
        return await this.evaluateCustom(condition, features);

      default:
        return false;
    }
  }

  private evaluateComparison(condition: RuleCondition, features: RequestFeatures): boolean {
    const { field, operator, value } = condition;
    if (!field || !operator) return false;

    const featureValue = this.getFeatureValue(features, field);

    switch (operator) {
      case 'eq': return featureValue === value;
      case 'ne': return featureValue !== value;
      case 'gt': return featureValue > value;
      case 'lt': return featureValue < value;
      case 'gte': return featureValue >= value;
      case 'lte': return featureValue <= value;
      case 'contains':
        return typeof featureValue === 'string' &&
               typeof value === 'string' &&
               featureValue.includes(value);
      default: return false;
    }
  }
}
```

```typescript
class BotDetectionModel {
  private model: tf.LayersModel;
  private featureEncoder: FeatureEncoder;
  private threshold: number = 0.7;

  async initialize(): Promise<void> {
    // 加载预训练模型
    this.model = await tf.loadLayersModel('file://./models/bot-detection/model.json');

    // 初始化特征编码器
    this.featureEncoder = new FeatureEncoder();
    await this.featureEncoder.initialize();
  }

  async predict(features: RequestFeatures): Promise<BotPrediction> {
    // 特征编码
    const encodedFeatures = await this.featureEncoder.encode(features);

    // 模型预测
    const inputTensor = tf.tensor2d([encodedFeatures]);
    const predictionTensor = this.model.predict(inputTensor) as tf.Tensor;
    const predictionArray = await predictionTensor.array() as number[][];

    // 清理TensorFlow内存
    inputTensor.dispose();
    predictionTensor.dispose();

    const botProbability = predictionArray[0][0];
    const isBot = botProbability >= this.threshold;

    return {
      isBot,
      probability: botProbability,
      confidence: Math.abs(botProbability - 0.5) * 2, // 距离0.5的绝对距离的两倍
      features: this.extractImportantFeatures(features, encodedFeatures)
    };
  }

  private extractImportantFeatures(
    originalFeatures: RequestFeatures,
    encodedFeatures: number[]
  ): ImportantFeature[] {
    // 使用SHAP或LIME方法计算特征重要性
    const featureImportance = this.calculateFeatureImportance(encodedFeatures);

    return Object.entries(featureImportance)
      .sort((a, b) => b[1] - a[1])
      .slice(0, 5)
      .map(([featureName, importance]) => ({
        name: featureName,
        importance,
        value: this.getFeatureValue(originalFeatures, featureName),
        contribution: importance * 100 // 百分比贡献度
      }));
  }
}

interface BotPrediction {
  isBot: boolean;
  probability: number;
  confidence: number;
  features: ImportantFeature[];
}

interface ImportantFeature {
  name: string;
  importance: number;
  value: any;
  contribution: number;
}
```

```typescript
class ProtectionStrategy {
  private strategies: Map<string, ProtectionAction> = new Map();

  constructor() {
    this.initializeStrategies();
  }

  private initializeStrategies(): void {
    // 轻度防护策略
    this.strategies.set('low', {

      }
    });

    // 中度防护策略
    this.strategies.set('medium', {
      type: 'challenge',
      parameters: {
        challengeType: 'captcha',
        difficulty: 'medium',
        expiration: 300 // 5分钟
      }
    });

    // 重度防护策略
    this.strategies.set('high', {
      type: 'temporary_block',
      parameters: {
        duration: 3600, // 1小时
        scope: 'ip',
        reason: 'suspicious_bot_activity'
      }
    });

    // 严重防护策略
    this.strategies.set('critical', {
      type: 'permanent_block',
      parameters: {
        scope: 'ip',
        reason: 'confirmed_bot_attack',
        notify: true
      }
    });
  }

  async applyProtection(
    threatLevel: 'low' | 'medium' | 'high' | 'critical',
    request: Request,
    detectionResult: DetectionResult
  ): Promise<ProtectionResult> {
    const strategy = this.strategies.get(threatLevel);
    if (!strategy) {
      return { applied: false, reason: 'no_strategy_found' };
    }

    try {
      switch (strategy.type) {
        case 'rate_limit':
          return await this.applyRateLimit(strategy.parameters, request);

        case 'challenge':
          return await this.applyChallenge(strategy.parameters, request);

        case 'temporary_block':
          return await this.applyTemporaryBlock(strategy.parameters, request);

        case 'permanent_block':
          return await this.applyPermanentBlock(strategy.parameters, request);

        default:
          return { applied: false, reason: 'unknown_strategy_type' };
      }
    } catch (error) {
      console.error('Error applying protection strategy:', error);
      return { applied: false, reason: 'application_error', error };
    }
  }

  private async applyRateLimit(
    parameters: RateLimitParameters,
    request: Request
  ): Promise<ProtectionResult> {
    const { requestsPerMinute, burstSize, scope } = parameters;
    const identifier = this.getIdentifier(request, scope);

    const key = `rate_limit:${identifier}`;
    const now = Date.now();
    const windowStart = now - 60 * 1000;

    // 使用Redis Lua脚本实现原子操作
    const luaScript = `
      local key = KEYS[1]
      local now = tonumber(ARGV[1])
      local windowStart = tonumber(ARGV[2])
      local limit = tonumber(ARGV[3])
      local burst = tonumber(ARGV[4])

      -- 移除过期请求
      redis.call('ZREMRANGEBYSCORE', key, 0, windowStart)

      -- 获取当前计数
      local current = redis.call('ZCARD', key)

      if current >= limit + burst then
        return {0, 0, 0} -- 超过限制
      end

      -- 添加当前请求
      redis.call('ZADD', key, now, now)
      redis.call('EXPIRE', key, 120)

      -- 计算剩余配额
      local remaining = math.max(0, limit + burst - current - 1)
      local resetTime = windowStart + 60000

      return {1, remaining, resetTime}
    `;

    const result = await redisClient.eval(
      luaScript,
      1, // KEYS数量
      key,
      now.toString(),
      windowStart.toString(),
      requestsPerMinute.toString(),
      burstSize.toString()
    ) as [number, number, number];

    const [allowed, remaining, resetTime] = result;

    return {
      applied: true,
      allowed: allowed === 1,
      details: {
        type: 'rate_limit',
        limit: requestsPerMinute,
        remaining,
        resetTime: new Date(resetTime),
        identifier
      }
    };
  }
}
```

```typescript
          results.push(detection);

          await this.recordDetection(honeypotId, detection);
        }
      }
    }

    return {
      triggered: results.length > 0,
      detections: results,
      threatLevel: this.calculateThreatLevel(results)
    };
  }

  private async checkHoneypot(
    config: HoneypotConfig,
    request: Request
  ): Promise<HoneypotDetection | null> {
    switch (config.type) {
      case 'form':
        return await this.checkFormHoneypot(config, request);

      case 'link':
        return await this.checkLinkHoneypot(config, request);

      case 'api':
        return await this.checkApiHoneypot(config, request);

      case 'timing':
        return await this.checkTimingHoneypot(config, request);

      default:
        return null;
    }
  }

  private async checkFormHoneypot(
    config: HoneypotConfig,
    request: Request
  ): Promise<HoneypotDetection | null> {
    if (request.method !== 'POST') return null;

    const honeypotValue = request.body[config.fieldName!];
    if (honeypotValue !== undefined && honeypotValue !== config.expectedValue) {
      return {
        honeypotId: config.id,
        type: config.type,
        triggered: true,
        evidence: {
          field: config.fieldName!,
          expected: config.expectedValue,
          actual: honeypotValue
        },
        timestamp: new Date(),
        confidence: 0.95
      };
    }

    return null;
  }

  private calculateThreatLevel(detections: HoneypotDetection[]): number {
    if (detections.length === 0) return 0;

    let totalConfidence = 0;
    for (const detection of detections) {
      totalConfidence += detection.confidence;
    }

    const averageConfidence = totalConfidence / detections.length;

    // 基于置信度和触发数量计算威胁级别
    const baseThreat = averageConfidence * 100;
    const quantityBonus = Math.min(detections.length * 10, 30); // 最多30%加成

    return Math.min(baseThreat + quantityBonus, 100);
  }
}
```

#### 3.2.1 滑动窗口算法优化

系统采用优化的滑动窗口算法，实现精确的速率限制：

```typescript
class OptimizedSlidingWindow {
  private windowSize: number; // 窗口大小（毫秒）
  private maxRequests: number; // 最大请求数
  private precision: number; // 精度（毫秒）

  constructor(windowSize: number, maxRequests: number, precision: number = 1000) {
    this.windowSize = windowSize;
    this.maxRequests = maxRequests;
    this.precision = precision;
  }

  async checkRateLimit(
    identifier: string,
    timestamp: number = Date.now()
  ): Promise<RateLimitResult> {
    const windowKey = `rate_limit:${identifier}:window`;
    const requestKey = `rate_limit:${identifier}:requests`;

    const windowStart = timestamp - this.windowSize;

    // 使用Redis Lua脚本实现原子操作
    const luaScript = `
      local windowKey = KEYS[1]
      local requestKey = KEYS[2]
      local timestamp = tonumber(ARGV[1])
      local windowStart = tonumber(ARGV[2])
      local windowSize = tonumber(ARGV[3])
      local maxRequests = tonumber(ARGV[4])
      local precision = tonumber(ARGV[5])

      -- 获取或创建当前时间窗口
      local currentWindow = redis.call('GET', windowKey)
      local windowId

      if not currentWindow then
        -- 创建新窗口
        windowId = math.floor(timestamp / precision)
        redis.call('SET', windowKey, windowId)
        redis.call('EXPIRE', windowKey, math.ceil(windowSize / 1000) + 10)
      else
        windowId = tonumber(currentWindow)
      end

      -- 计算时间片
      local sliceId = math.floor(timestamp / precision)

      -- 如果时间片超出当前窗口，创建新窗口
      if sliceId > windowId then
        -- 删除旧窗口数据
        redis.call('DEL', requestKey)

        -- 创建新窗口
        windowId = sliceId
        redis.call('SET', windowKey, windowId)
        redis.call('EXPIRE', windowKey, math.ceil(windowSize / 1000) + 10)
      end

      -- 添加当前请求
      local requestId = tostring(timestamp) .. ':' .. tostring(math.random())
      redis.call('ZADD', requestKey, timestamp, requestId)

      -- 移除过期请求
      redis.call('ZREMRANGEBYSCORE', requestKey, 0, windowStart)

      -- 设置请求集合过期时间
      redis.call('EXPIRE', requestKey, math.ceil(windowSize / 1000) + 10)

      -- 计算当前请求数
      local requestCount = redis.call('ZCOUNT', requestKey, windowStart, timestamp)

      -- 检查是否超过限制
      local allowed = requestCount <= maxRequests
      local remaining = math.max(0, maxRequests - requestCount)
      local resetTime = windowStart + windowSize

      return {allowed and 1 or 0, remaining, resetTime}
    `;

    const result = await redisClient.eval(
      luaScript,
      2, // KEYS数量
      windowKey,
      requestKey,
      timestamp.toString(),
      windowStart.toString(),
      this.windowSize.toString(),
      this.maxRequests.toString(),
      this.precision.toString()
    ) as [number, number, number];

    const [allowed, remaining, resetTime] = result;

    return {
      allowed: allowed === 1,
      remaining,
      resetTime: new Date(resetTime),
      limit: this.maxRequests,
      windowSize: this.windowSize
    };
  }
}
```

#### 3.2.2 动态配额调整算法

系统根据历史数据和实时负载动态调整速率限制配额：

```typescript
class DynamicQuotaAdjuster {
  private history: RateLimitHistory[] = [];
  private readonly maxHistorySize = 1000;
  private readonly learningRate = 0.1;

  async adjustQuota(
    currentQuota: number,
    currentLoad: SystemLoad,
    historicalData: HistoricalStats
  ): Promise<AdjustedQuota> {
    // 基于多个因素计算调整系数
    const factors = await this.calculateAdjustmentFactors(currentLoad, historicalData);

    // 加权计算总调整系数
    const totalAdjustment = this.calculateTotalAdjustment(factors);

    // 应用调整，确保在合理范围内
    const newQuota = Math.max(
      1, // 最小配额
      Math.min(
        10000, // 最大配额
        Math.round(currentQuota * totalAdjustment)
      )
    );

    // 记录调整历史
    await this.recordAdjustment({
      timestamp: new Date(),
      oldQuota: currentQuota,
      newQuota,
      factors,
      totalAdjustment,
      currentLoad,
      reason: this.generateAdjustmentReason(factors)
    });

    return {
      quota: newQuota,
      adjustment: totalAdjustment,
      factors,
      effectiveFrom: new Date(),
      expiresAt: new Date(Date.now() + 5 * 60 * 1000) // 5分钟后重新评估
    };
  }

  private async calculateAdjustmentFactors(
    currentLoad: SystemLoad,
    historicalData: HistoricalStats
  ): Promise<AdjustmentFactors> {
    const factors: AdjustmentFactors = {
      loadFactor: 1.0,
      timeFactor: 1.0,
      userFactor: 1.0,
      errorFactor: 1.0,
      seasonalFactor: 1.0
    };

    // 1. 基于系统负载的调整
    factors.loadFactor = this.calculateLoadFactor(currentLoad);

    // 2. 基于时间的调整（考虑高峰时段）
    factors.timeFactor = this.calculateTimeFactor();

    // 3. 基于用户行为的调整
    factors.userFactor = await this.calculateUserFactor(historicalData);

    // 4. 基于错误率的调整
    factors.errorFactor = this.calculateErrorFactor(historicalData);

    // 5. 基于季节性的调整
    factors.seasonalFactor = await this.calculateSeasonalFactor();

    return factors;
  }

  private calculateLoadFactor(currentLoad: SystemLoad): number {
    const { cpuUsage, memoryUsage, requestRate } = currentLoad;

    // 综合负载评分（0-1）
    const loadScore = (
      cpuUsage * 0.4 +
      memoryUsage * 0.3 +
      Math.min(requestRate / 1000, 1) * 0.3
    );

    // 负载越高，配额越低
    if (loadScore < 0.3) return 1.2;  // 低负载，增加配额
    if (loadScore < 0.7) return 1.0;  // 正常负载，保持配额
    if (loadScore < 0.9) return 0.8;  // 高负载，减少配额
    return 0.5;  // 过载，大幅减少配额
  }

  private calculateTimeFactor(): number {
    const now = new Date();
    const hour = now.getHours();
    const day = now.getDay(); // 0=周日, 6=周六

    // 工作日高峰时段（9-18点）
    if (day >= 1 && day <= 5) {
      if (hour >= 9 && hour < 12) return 0.8;  // 上午高峰
      if (hour >= 14 && hour < 18) return 0.8; // 下午高峰
    }

    // 周末
    if (day === 0 || day === 6) {
      if (hour >= 20 && hour < 24) return 0.9; // 晚间使用
    }

    return 1.0; // 正常时段
  }

  private calculateTotalAdjustment(factors: AdjustmentFactors): number {
    // 使用几何平均计算总调整系数
    const values = [
      factors.loadFactor,
      factors.timeFactor,
      factors.userFactor,
      factors.errorFactor,
      factors.seasonalFactor
    ];

    const product = values.reduce((acc, val) => acc * val, 1);
    const geometricMean = Math.pow(product, 1 / values.length);

    // 应用平滑处理，避免剧烈变化
    return this.applySmoothing(geometricMean);
  }

  private applySmoothing(value: number): number {
    // 使用指数移动平均进行平滑
    const alpha = 0.3; // 平滑系数
    const lastValue = this.history.length > 0
      ? this.history[this.history.length - 1].totalAdjustment
      : value;

    return alpha * value + (1 - alpha) * lastValue;
  }
}
```
